```javascript
// === src/services/virtualScroller.pool.js ===
// Vanilla ES6+ | Zero Dependencies | iPad A10 Optimized | 3-Node Pool Max

export class CodexVirtualPool {
  constructor({
    viewportEl,           // Contenedor scrollable
    totalChunks,          // Número total de chunks
    getHeightForChunk,    // Función: (chunkId) => height en px
    renderChunk,          // Función: (chunkId, poolNode) => Promise<void>
    overscan = 0,         //Chunks adicionales (0 para low-end)
    poolSize = 3,         // NÚMERO SAGRADO: 3 nodos máximo
    throttleMs = 16       // Mínimo entre cálculos (1 frame @60fps)
  }) {
    // 🔥 Validación brutal: si poolSize > 3 en low-end, corregir
    const isLowEnd = navigator.deviceMemory <= 2 || 
                    /iPad.*CPU OS 1[0-4]_/.test(navigator.userAgent);
    if (isLowEnd && poolSize > 3) {
      console.warn('⚠️ Forzando poolSize=3 para iPad A10');
      poolSize = 3;
    }
    
    this.viewport = viewportEl;
    this.totalChunks = totalChunks;
    this.getHeight = getHeightForChunk;
    this.render = renderChunk;
    this.overscan = isLowEnd ? 0 : overscan;
    this.poolSize = poolSize;
    this.throttleMs = isLowEnd ? 32 : throttleMs; // 30fps target en low-end
    
    // Estado interno
    this.pool = [];                    // Array de nodos disponibles {el, chunkId, busy}
    this.activeMap = new Map();        // chunkId → poolNode reference
    this.heightMap = new Float64Array(totalChunks); // Altura acumulada por chunk
    this.offsetTopMap = new Float64Array(totalChunks); // Posición Y absoluta por chunk
    this.visibleRange = { start: -1, end: -1 };
    
    // Control de ciclo de vida
    this.lastScrollCalc = 0;
    this.rafId = null;
    this.scrollY = 0;
    this.viewportHeight = 0;
    this.isScrolling = false;
    this.scrollTimeout = null;
    
    // Métricas para debugging
    this.metrics = {
      renders: 0,
      recycles: 0,
      drops: 0,
      lastFrameTime: 0
    };
  }

  // === FASE 1: PRE-COMPUTAR MAPA DE ALTURAS (una vez, al init) ===
  async precomputeHeights() {
    // Calcular offset acumulativo para cada chunk: O(n) pero solo una vez
    let cumulative = 0;
    for (let i = 0; i < this.totalChunks; i++) {
      this.offsetTopMap[i] = cumulative;
      const h = await this.getHeight(i);
      this.heightMap[i] = h;
      cumulative += h;
    }
    // Altura total del viewport fantasma
    this.viewport.style.height = `${cumulative}px`;
    this.viewport.style.contain = 'strict'; // Crucial: aislar repaints
    console.log(`📐 Height map computed: ${cumulative}px total`);
  }

  // === FASE 2: INICIALIZAR POOL DE 3 NODOS ===
  initPool() {
    // Crear exactamente `poolSize` nodos reutilizables
    for (let i = 0; i < this.poolSize; i++) {
      const node = document.createElement('div');
      node.className = 'virtual-pool-node';
      node.style.position = 'absolute';
      node.style.left = '0';
      node.style.right = '0';
      node.style.top = '0'; // Se ajustará con transform
      node.style.willChange = 'transform';
      node.style.contain = 'content'; // Aislar layout interno
      node.setAttribute('aria-busy', 'false');
      this.pool.push({ el: node, chunkId: null, busy: false });
    }
    // Añadir todos al viewport (ocultos hasta que se usen)
    this.pool.forEach(({ el }) => {
      el.style.opacity = '0';
      el.style.pointerEvents = 'none';
      this.viewport.appendChild(el);
    });
    console.log(`🏊 Pool initialized: ${this.poolSize} nodes`);
  }

  // === FASE 3: MOUNT + SCROLL LISTENER CON THROTTLE ===
  mount() {
    this.viewportHeight = this.viewport.clientHeight;
    this._updateViewportHeight = this._updateViewportHeight.bind(this);
    window.addEventListener('resize', this._updateViewportHeight, { passive: true });
    
    // Scroll listener con passive: true + throttle manual
    this._onScroll = this._onScroll.bind(this);
    this.viewport.addEventListener('scroll', this._onScroll, { passive: true });
    
    // Initial render del rango visible
    this._scheduleRecalc();
    
    return () => this.destroy();
  }

  _updateViewportHeight() {
    this.viewportHeight = this.viewport.clientHeight;
    this._scheduleRecalc();
  }

  _onScroll() {
    this.scrollY = this.viewport.scrollTop;
    this.isScrolling = true;
    
    // Throttle: solo recalcular si pasó throttleMs desde el último cálculo
    const now = performance.now();
    if (now - this.lastScrollCalc >= this.throttleMs) {
      this._scheduleRecalc();
      this.lastScrollCalc = now;
    }
    
    // Detectar fin de scroll para cleanup diferido
    clearTimeout(this.scrollTimeout);
    this.scrollTimeout = setTimeout(() => {
      this.isScrolling = false;
      this._cleanupOffscreen();
    }, 150);
  }

  // === FASE 4: SCHEDULE CON requestAnimationFrame ===
  _scheduleRecalc() {
    if (this.rafId !== null) return; // Ya hay un frame pendiente
    
    this.rafId = requestAnimationFrame(() => {
      this.rafId = null;
      this._recalcVisibleRange();
    });
  }

  // === FASE 5: CÁLCULO DE RANGO VISIBLE + RECYCLING ===
  _recalcVisibleRange() {
    const start = this._findChunkAtY(this.scrollY - (this.overscan * 2000)); // Overscan conservador
    const end = this._findChunkAtY(this.scrollY + this.viewportHeight + (this.overscan * 2000));
    
    // Si el rango no cambió, abortar
    if (start === this.visibleRange.start && end === this.visibleRange.end) return;
    
    const prevVisible = new Set(this.activeMap.keys());
    const newVisible = new Set();
    
    // 1. Marcar chunks que deben estar visibles
    for (let i = start; i <= end; i++) {
      if (i < 0 || i >= this.totalChunks) continue;
      newVisible.add(i);
      
      // Si ya está renderizado, solo ajustar posición
      if (this.activeMap.has(i)) {
        this._updateNodePosition(i);
      }
    }
    
    // 2. Liberar chunks que salieron del rango + overscan
    for (const chunkId of prevVisible) {
      if (!newVisible.has(chunkId)) {
        this._releaseNode(chunkId);
      }
    }
    
    // 3. Renderizar chunks nuevos que entraron (con prioridad al centro del viewport)
    const toRender = [];
    for (const chunkId of newVisible) {
      if (!this.activeMap.has(chunkId)) {
        toRender.push(chunkId);
      }
    }
    
    // Ordenar por proximidad al centro del viewport (prioridad visual)
    const viewportCenter = this.scrollY + this.viewportHeight / 2;
    toRender.sort((a, b) => {
      const aCenter = this.offsetTopMap[a] + this.heightMap[a] / 2;
      const bCenter = this.offsetTopMap[b] + this.heightMap[b] / 2;
      return Math.abs(aCenter - viewportCenter) - Math.abs(bCenter - viewportCenter);
    });
    
    // 4. Renderizar con límite estricto de poolSize
    let rendered = 0;
    for (const chunkId of toRender) {
      if (rendered >= this.poolSize) {
        // Pool lleno: registrar drop para métricas
        this.metrics.drops++;
        break;
      }
      this._acquireAndRender(chunkId);
      rendered++;
    }
    
    // Actualizar rango visible
    this.visibleRange = { start, end };
    
    // Registrar métrica de frame
    const now = performance.now();
    const frameTime = now - this.metrics.lastFrameTime;
    this.metrics.lastFrameTime = now;
    if (frameTime > 33) {
      console.warn(`⚠️ Frame lento: ${frameTime.toFixed(1)}ms (chunk range: ${start}-${end})`);
    }
  }

  // === FASE 6: BÚSQUEDA BINARIA EN HEIGHT MAP (O(log n)) ===
  _findChunkAtY(y) {
    // Clamp a límites válidos
    if (y < 0) return 0;
    if (y >= this.offsetTopMap[this.totalChunks - 1] + this.heightMap[this.totalChunks - 1]) {
      return this.totalChunks - 1;
    }
    
    // Búsqueda binaria en offsetTableMap (array ordenado)
    let lo = 0, hi = this.totalChunks - 1;
    while (lo < hi) {
      const mid = (lo + hi + 1) >> 1; // Mid superior para evitar bucle infinito
      if (this.offsetTopMap[mid] <= y) {
        lo = mid;
      } else {
        hi = mid - 1;
      }
    }
    return lo;
  }

  // === FASE 7: ADQUISICIÓN DE NODO DEL POOL + RENDER ===
  async _acquireAndRender(chunkId) {
    // Buscar nodo disponible en el pool
    const node = this.pool.find(n => !n.busy);
    if (!node) {
      // Pool lleno: esto no debería pasar si el límite de renders se respeta
      console.warn(`⚠️ Pool exhausted trying to render chunk ${chunkId}`);
      return;
    }
    
    // Marcar como ocupado
    node.busy = true;
    node.chunkId = chunkId;
    this.activeMap.set(chunkId, node);
    
    // Posicionar inicialmente fuera de vista para evitar FOUC
    node.el.style.transform = `translateY(${this.offsetTopMap[chunkId]}px)`;
    node.el.setAttribute('data-chunk', chunkId);
    node.el.setAttribute('aria-busy', 'true');
    
    // Renderizar contenido (función proporcionada por el caller)
    try {
      const start = performance.now();
      await this.render(chunkId, node.el);
      const renderTime = performance.now() - start;
      
      // Mostrar nodo con fade-in sutil
      node.el.style.transition = 'opacity 0.1s ease-out';
      node.el.style.opacity = '1';
      node.el.style.pointerEvents = 'auto';
      node.el.setAttribute('aria-busy', 'false');
      
      // Métricas
      this.metrics.renders++;
      if (renderTime > 16) {
        console.warn(`⚠️ Render lento chunk ${chunkId}: ${renderTime.toFixed(1)}ms`);
      }
    } catch (e) {
      console.error(`❌ Error rendering chunk ${chunkId}:`, e);
      this._renderErrorPlaceholder(node.el, chunkId);
    }
  }

  // === FASE 8: LIBERACIÓN DE NODO (RECICLAJE) ===
  _releaseNode(chunkId) {
    const node = this.activeMap.get(chunkId);
    if (!node) return;
    
    // Ocultar inmediatamente para evitar repaint innecesario
    node.el.style.opacity = '0';
    node.el.style.pointerEvents = 'none';
    node.el.setAttribute('aria-busy', 'false');
    
    // Limpiar contenido para liberar memoria (crítico en 2GB RAM)
    // Usar replaceChildren() si está disponible, fallback a innerHTML = ''
    if (node.el.replaceChildren) {
      node.el.replaceChildren();
    } else {
      node.el.innerHTML = '';
    }
    
    // Resetear estado del nodo
    node.busy = false;
    node.chunkId = null;
    this.activeMap.delete(chunkId);
    
    // Métrica de reciclaje
    this.metrics.recycles++;
  }

  // === FASE 9: ACTUALIZAR POSICIÓN DE NODO YA RENDERIZADO ===
  _updateNodePosition(chunkId) {
    const node = this.activeMap.get(chunkId);
    if (!node) return;
    
    // Solo actualizar transform si la posición cambió significativamente
    const targetY = this.offsetTopMap[chunkId];
    const currentTransform = node.el.style.transform;
    
    // Parsear valor actual (optimización: evitar reflow leyendo style)
    const match = currentTransform?.match(/translateY\(([\d.]+)px\)/);
    const currentY = match ? parseFloat(match[1]) : null;
    
    // Threshold de 1px para evitar updates innecesarios
    if (currentY === null || Math.abs(currentY - targetY) > 1) {
      node.el.style.transform = `translateY(${targetY}px)`;
    }
  }

  // === FASE 10: CLEANUP DIFERIDO POST-SCROLL ===
  _cleanupOffscreen() {
    // Solo limpiar si no estamos en scroll activo
    if (this.isScrolling) return;
    
    // Liberar chunks fuera del rango visible + buffer generoso
    const buffer = this.viewportHeight * 2;
    for (const [chunkId, node] of this.activeMap) {
      const chunkTop = this.offsetTopMap[chunkId];
      const chunkBottom = chunkTop + this.heightMap[chunkId];
      
      const isOffscreen = chunkBottom < this.scrollY - buffer || 
                         chunkTop > this.scrollY + this.viewportHeight + buffer;
      
      if (isOffscreen) {
        this._releaseNode(chunkId);
      }
    }
  }

  // === UTILS ===
  _renderErrorPlaceholder(el, chunkId) {
    el.innerHTML = `<div class="chunk-error" style="height:${this.heightMap[chunkId]}px;display:flex;align-items:center;justify-content:center;background:#fff3f3;color:#dc2626">⚠️ Error cargando sección</div>`;
    el.style.opacity = '1';
    el.style.pointerEvents = 'auto';
  }

  scrollToChunk(chunkId, offset = 0) {
    const targetY = this.offsetTopMap[chunkId] + offset;
    this.viewport.scrollTo({ top: targetY, behavior: 'smooth' });
  }

  getMetrics() {
    return { ...this.metrics, activeNodes: this.activeMap.size };
  }

  destroy() {
    if (this.rafId) cancelAnimationFrame(this.rafId);
    clearTimeout(this.scrollTimeout);
    window.removeEventListener('resize', this._updateViewportHeight);
    this.viewport.removeEventListener('scroll', this._onScroll);
    
    // Limpiar pool
    this.pool.forEach(({ el }) => el.remove());
    this.pool = [];
    this.activeMap.clear();
    
    console.log('🗑️ VirtualPool destroyed');
  }
}
```

```javascript
// === src/main.codex.js ===
import { CodexVirtualPool } from './services/virtualScroller.pool.js';
import { CodexOfflineStorage } from './services/offlineStorage.js';

async function initCodex() {
  const viewport = document.getElementById('codex-viewport');
  const TOTAL_CHUNKS = 400; // Ejemplo: 400 chunks de ~50 artículos cada uno
  
  // 1. Inicializar almacenamiento offline híbrido
  const storage = new CodexOfflineStorage();
  await storage.init();
  
  // 2. Función para obtener altura de chunk (estimada + medición real post-render)
  const chunkHeights = new Map();
  const getHeightForChunk = async (chunkId) => {
    // Si ya medimos este chunk, devolver valor cacheado
    if (chunkHeights.has(chunkId)) return chunkHeights.get(chunkId);
    
    // Estimación conservadora: 1200px promedio, clamp entre 200-4000px
    const estimated = 1200;
    return Math.max(200, Math.min(4000, estimated));
  };
  
  // 3. Función de renderizado: carga chunk + inyecta + mide altura real
  const renderChunk = async (chunkId, containerEl) => {
    // Obtener HTML del chunk (IDB → Cache → Red)
    const { html } = await storage.getChunk(chunkId);
    
    // Inyectar de forma segura
    containerEl.innerHTML = html;
    
    // Medir altura real post-render (para refinar heightMap)
    await new Promise(resolve => {
      // Usar requestIdleCallback para no bloquear el paint
      if ('requestIdleCallback' in window) {
        requestIdleCallback(() => {
          const realHeight = containerEl.offsetHeight;
          chunkHeights.set(chunkId, realHeight);
          resolve();
        }, { timeout: 100 });
      } else {
        setTimeout(() => {
          const realHeight = containerEl.offsetHeight;
          chunkHeights.set(chunkId, realHeight);
          resolve();
        }, 0);
      }
    });
    
    // Highlighting diferido (solo si está cerca del viewport)
    if (window.hljs?.highlightElement) {
      // Usar Web Worker si está disponible, fallback a idle callback
      if ('Worker' in window) {
        // ... lógica de worker para highlight ...
      } else {
        if ('requestIdleCallback' in window) {
          requestIdleCallback(() => {
            containerEl.querySelectorAll('pre code').forEach(hljs.highlightElement);
          }, { timeout: 2000 });
        }
      }
    }
  };
  
  // 4. Instanciar VirtualPool con parámetros agresivos para iPad A10
  const pool = new CodexVirtualPool({
    viewportEl: viewport,
    totalChunks: TOTAL_CHUNKS,
    getHeightForChunk,
    renderChunk,
    overscan: 0,           // Cero precarga en low-end
    poolSize: 3,           // SAGRADO: máximo 3 nodos en DOM
    throttleMs: 32         // Target 30fps para reducir carga térmica
  });
  
  // 5. Pre-computar alturas (una vez)
  await pool.precomputeHeights();
  
  // 6. Inicializar pool y montar
  pool.initPool();
  pool.mount();
  
  // 7. Exponer para debugging (solo dev)
  if (import.meta.env?.DEV) {
    window.__codexPool = pool;
    window.__codexStorage = storage;
  }
  
  return pool;
}

// Boot
```

```javascript
// === src/services/perfMonitor.ipad.js ===
// Vanilla ES6+ | Safari 14.1 Compatible | iPad A10 Thermal Awareness

export class CodexPerfMonitor {
  constructor({
    thermalThresholdCelsius = 75,  // Temperatura para activar modo throttle
    fpsTarget = 30,                 // FPS mínimo aceptable en low-end
    gcWarningThreshold = 50,        // MB de GC por segundo que disparan alerta
    sampleIntervalMs = 1000         // Frecuencia de muestreo
  } = {}) {
    this.thermalThreshold = thermalThresholdCelsius;
    this.fpsTarget = fpsTarget;
    this.gcThreshold = gcWarningThreshold;
    this.sampleInterval = sampleIntervalMs;
    
    // Estado de monitoreo
    this.isMonitoring = false;
    this.thermalMode = false;
    this.lastFrameTime = performance.now();
    this.frameCount = 0;
    this.fpsHistory = [];
    this.gcHistory = [];
    
    // Callbacks para reacción externa
    this.onThermalEnter = null;
    this.onThermalExit = null;
    this.onFPSDrop = null;
    this.onGCPressure = null;
    
    // Feature detection para Safari 14
    this.supportsPerformanceObserver = 'PerformanceObserver' in window;
    this.supportsMemory = 'memory' in performance; // Solo Chrome/Edge, NO Safari
    this.supportsThermal = 'getTemperature' in navigator; // NO existe en Safari, fakeamos
    
    // Fallbacks para Safari
    this._rafDeltas = [];
    this._lastGCGuess = null;
  }

  // === INICIAR MONITOREO ===
  start() {
    if (this.isMonitoring) return;
    this.isMonitoring = true;
    
    console.log('🔍 CodexPerfMonitor started (Safari 14 mode)');
    
    // 1. FPS monitoring via requestAnimationFrame deltas
    this._monitorFPS();
    
    // 2. Layout Thrashing detection (forced reflow tracking)
    this._monitorLayoutThrashing();
    
    // 3. GC pressure estimation (heuristic para Safari)
    this._monitorGCEstimate();
    
    // 4. Thermal estimation (proxy-based para Safari)
    this._monitorThermalProxy();
    
    // 5. PerformanceObserver para long tasks (si está disponible)
    if (this.supportsPerformanceObserver) {
      this._initLongTaskObserver();
    }
  }

  // === FPS MONITORING (requestAnimationFrame deltas) ===
  _monitorFPS() {
    const measureFrame = (timestamp) => {
      if (!this.isMonitoring) return;
      
      const delta = timestamp - this.lastFrameTime;
      this.lastFrameTime = timestamp;
      this.frameCount++;
      
      // Registrar delta para análisis
      this._rafDeltas.push(delta);
      if (this._rafDeltas.length > 60) this._rafDeltas.shift(); // Últimos 60 frames (~1s @60fps)
      
      // Calcular FPS instantáneo
      const instantFPS = 1000 / delta;
      this.fpsHistory.push(instantFPS);
      if (this.fpsHistory.length > 30) this.fpsHistory.shift(); // Últimos 30 muestras
      
      // Detectar drop de FPS
      const avgFPS = this.fpsHistory.reduce((a,b) => a+b, 0) / this.fpsHistory.length;
      if (avgFPS < this.fpsTarget && !this.thermalMode) {
        this._triggerFPSDrop(avgFPS);
      }
      
      // Programar siguiente frame
      requestAnimationFrame(measureFrame);
    };
    
    requestAnimationFrame(measureFrame);
  }

  _triggerFPSDrop(avgFPS) {
    console.warn(`⚠️ FPS drop detected: ${avgFPS.toFixed(1)}fps < target ${this.fpsTarget}fps`);
    if (this.onFPSDrop) this.onFPSDrop(avgFPS);
    
    // Si el drop es severo, activar modo térmico preventivo
    if (avgFPS < this.fpsTarget * 0.7) {
      this._enterThermalMode('fps_drop');
    }
  }

  // === LAYOUT THRASHING DETECTION ===
  _monitorLayoutThrashing() {
    // Heurística: contar lecturas de layout-forcing properties dentro de un frame
    // Properties que fuerzan reflow: offsetHeight, scrollTop, getBoundingClientRect(), etc.
    
    const layoutForcingProps = [
      'offsetHeight', 'offsetWidth', 'clientHeight', 'clientWidth',
      'scrollHeight', 'scrollWidth', 'scrollTop', 'scrollLeft',
      'getBoundingClientRect', 'getClientRects'
    ];
    
    let readCount = 0;
    let writeCount = 0;
    let lastFrameReads = 0;
    
    // Monkey-patch para detectar accesos (solo en modo dev, con flag)
    if (import.meta.env?.DEV) {
      const originalGetComputedStyle = window.getComputedStyle;
      window.getComputedStyle = function(...args) {
        readCount++;
        return originalGetComputedStyle.apply(this, args);
      };
      
      // Interceptar style writes
      const originalSetProperty = CSSStyleDeclaration.prototype.setProperty;
      CSSStyleDeclaration.prototype.setProperty = function(prop, value, priority) {
        writeCount++;
        return originalSetProperty.call(this, prop, value, priority);
      };
    }
    
    // Muestrear cada segundo
    setInterval(() => {
      if (!this.isMonitoring) return;
      
      // Detectar patrón de thrashing: muchas lecturas seguidas de muchas escrituras
      const readsThisFrame = readCount - lastFrameReads;
      lastFrameReads = readCount;
      
      // Threshold empírico para iPad A10: >50 layout reads por segundo = riesgo
      if (readsThisFrame > 50) {
        console.warn(`⚠️ Layout thrashing risk: ${readsThisFrame} forced reflows/s`);
        // Aquí podríamos activar modo "layout freeze" temporal
      }
      
      // Reset contadores
      readCount = 0;
      writeCount = 0;
    }, this.sampleInterval);
  }

  // === GC PRESSURE ESTIMATION (HEURÍSTICA PARA SAFARI 14) ===
  _monitorGCEstimate() {
    // Safari NO expone performance.memory, así que usamos heurísticas:
    // 1. Crecimiento de DOM nodes
    // 2. Frecuencia de creación/destrucción de objetos grandes
    // 3. Tiempo de respuesta de setTimeout (GC pausas bloquean el event loop)
    
    let lastNodeCount = document.querySelectorAll('*').length;
    let setTimeoutDrift = [];
    
    setInterval(() => {
      if (!this.isMonitoring) return;
      
      // 1. Medir crecimiento de DOM
      const currentNodeCount = document.querySelectorAll('*').length;
      const nodeDelta = currentNodeCount - lastNodeCount;
      lastNodeCount = currentNodeCount;
      
      // 2. Medir drift de setTimeout (GC pausa el event loop)
      const start = performance.now();
      setTimeout(() => {
        const drift = performance.now() - start - 0; // Esperábamos ~0ms de delay
        setTimeoutDrift.push(drift);
        if (setTimeoutDrift.length > 10) setTimeoutDrift.shift();
        
        const avgDrift = setTimeoutDrift.reduce((a,b) => a+b, 0) / setTimeoutDrift.length;
        
        // Si el drift promedio > 50ms, probable pausa de GC
        if (avgDrift > 50) {
          console.warn(`⚠️ GC pressure suspected: setTimeout drift ${avgDrift.toFixed(1)}ms`);
          if (this.onGCPressure) this.onGCPressure({ estimated: true, drift: avgDrift });
        }
      }, 0);
      
      // 3. Heurística combinada: muchos nodos nuevos + drift alto = GC inminente
      if (nodeDelta > 100 && setTimeoutDrift.length > 5) {
        const avgDrift = setTimeoutDrift.reduce((a,b) => a+b, 0) / setTimeoutDrift.length;
        if (avgDrift > 30) {
          console.warn(`⚠️ High GC risk: +${nodeDelta} nodes, drift ${avgDrift.toFixed(1)}ms`);
        }
      }
      
    }, this.sampleInterval);
  }

  // === THERMAL ESTIMATION (PROXY-BASED PARA SAFARI) ===
  _monitorThermalProxy() {
    // Safari NO expone temperatura CPU, así que inferimos por:
    // 1. FPS sostenidamente bajos a pesar de carga ligera
    // 2. Aumento de tiempo de ejecución de tareas simples
    // 3. Reducción de frecuencia de requestAnimationFrame callbacks
    
    let baselineTaskTime = null;
    let thermalScore = 0; // 0-100, donde 100 = throttling severo
    
    // Medir tiempo de tarea de referencia (operación trivial)
    const measureBaseline = () => {
      const start = performance.now();
      // Operación trivial pero no optimizable por JIT
      let sum = 0;
      for (let i = 0; i < 1000; i++) sum += Math.random();
      return performance.now() - start;
    };
    
    // Establecer baseline en los primeros 5 segundos
    setTimeout(() => {
      if (!baselineTaskTime) {
        const samples = [];
        for (let i = 0; i < 10; i++) samples.push(measureBaseline());
        baselineTaskTime = samples.reduce((a,b) => a+b, 0) / samples.length;
        console.log(`🌡️ Thermal baseline established: ${baselineTaskTime.toFixed(2)}ms`);
      }
    }, 5000);
    
    // Monitoreo continuo
    setInterval(() => {
      if (!this.isMonitoring || !baselineTaskTime) return;
      
      const currentTime = measureBaseline();
      const slowdownRatio = currentTime / baselineTaskTime;
      
      // Actualizar thermal score con suavizado exponencial
      thermalScore = thermalScore * 0.9 + (slowdownRatio - 1) * 100 * 0.1;
      thermalScore = Math.max(0, Math.min(100, thermalScore));
      
      // Detectar entrada/salida de modo térmico
      if (thermalScore > 70 && !this.thermalMode) {
        this._enterThermalMode('proxy_inference');
      } else if (thermalScore < 30 && this.thermalMode) {
        this._exitThermalMode();
      }
      
    }, this.sampleInterval);
  }

  _enterThermalMode(reason) {
    console.warn(`🔥 Entering thermal mode: ${reason}`);
    this.thermalMode = true;
    if (this.onThermalEnter) this.onThermalMode({ reason, timestamp: Date.now() });
  }

  _exitThermalMode() {
    console.log('❄️ Exiting thermal mode');
    this.thermalMode = false;
    if (this.onThermalExit) this.onThermalExit({ timestamp: Date.now() });
  }

  // === LONG TASK OBSERVER (si está disponible) ===
  _initLongTaskObserver() {
    try {
      const observer = new PerformanceObserver((list) => {
        for (const entry of list.getEntries()) {
          if (entry.duration > 50) {
            console.warn(`⚠️ Long task: ${entry.duration.toFixed(1)}ms`, entry.name);
          }
        }
      });
      observer.observe({ entryTypes: ['longtask'] });
    } catch (e) {
      // Safari 14 no soporta 'longtask', ignorar silenciosamente
    }
  }

  // === API PÚBLICA ===
  
  // Obtener métricas consolidadas
  getMetrics() {
    const avgFPS = this.fpsHistory.length > 0 
      ? this.fpsHistory.reduce((a,b) => a+b, 0) / this.fpsHistory.length 
      : null;
    
    return {
      thermalMode: this.thermalMode,
      fps: {
        current: this.fpsHistory[this.fpsHistory.length - 1] || null,
        average: avgFPS,
        history: [...this.fpsHistory]
      },
      rafDeltas: {
        average: this._rafDeltas.length > 0
          ? this._rafDeltas.reduce((a,b) => a+b, 0) / this._rafDeltas.length
          : null,
        p95: this._rafDeltas.length > 0
          ? [...this._rafDeltas].sort((a,b) => a-b)[Math.floor(this._rafDeltas.length * 0.95)]
          : null
      },
      // GC estimado (solo heurística en Safari)
      gc: {
        estimated: true,
        note: 'Safari 14 does not expose performance.memory'
      }
    };
  }
  
  // Callbacks para reacción externa
  setCallbacks({ onThermalEnter, onThermalExit, onFPSDrop, onGCPressure }) {
    this.onThermalEnter = onThermalEnter;
    this.onThermalExit = onThermalExit;
    this.onFPSDrop = onFPSDrop;
    this.onGCPressure = onGCPressure;
  }
  
  // Detener monitoreo
  stop() {
    this.isMonitoring = false;
    console.log('🛑 CodexPerfMonitor stopped');
  }
}
```

```javascript
// Conecta PerfMonitor con VirtualPool para reacción automática

export class CodexAdaptiveThrottle {
  constructor(virtualPool, perfMonitor) {
    this.pool = virtualPool;
    this.monitor = perfMonitor;
    
    // Estado de adaptación
    this.currentThrottleMs = 16; // Base: 60fps
    this.currentPoolSize = 3;    // Base: 3 nodos
    this.currentOverscan = 0;
    
    // Configurar callbacks del monitor
    this.monitor.setCallbacks({
      onThermalEnter: () => this._activateThermalMode(),
      onThermalExit: () => this._deactivateThermalMode(),
      onFPSDrop: (fps) => this._respondToFPSDrop(fps),
      onGCPressure: () => this._respondToGCPressure()
    });
  }
  
  _activateThermalMode() {
    console.log('🔥 AdaptiveThrottle: thermal mode ACTIVATED');
    
    // Reducir agresivamente la carga
    this.currentThrottleMs = 64;  // Target 15fps (mínimo aceptable)
    this.currentOverscan = 0;     // Cero precarga
    // Nota: NO reducimos poolSize de 3, es el mínimo para evitar blank screens
    
    // Aplicar cambios al pool (si el pool expone API de reconfiguración)
    if (this.pool.setThrottle) {
      this.pool.setThrottle(this.currentThrottleMs);
    }
  }
  
  _deactivateThermalMode() {
    console.log('❄️ AdaptiveThrottle: thermal mode DEACTIVATED');
    
    // Restaurar gradualmente
    this.currentThrottleMs = 32;  // Target 30fps para low-end
    this.currentOverscan = 0;     // Mantener cero en iPad A10 por seguridad
  }
  
  _respondToFPSDrop(fps) {
    if (fps < 20) {
      // Drop severo: aumentar throttle inmediatamente
      this.currentThrottleMs = Math.min(64, this.currentThrottleMs + 16);
      console.log(`⚠️ FPS ${fps.toFixed(1)}: throttle increased to ${this.currentThrottleMs}ms`);
    } else if (fps < 30) {
      // Drop moderado: ajuste suave
      this.currentThrottleMs = Math.min(48, this.currentThrottleMs + 8);
    }
  }
  
  _respondToGCPressure() {
    // GC pressure: reducir temporalmente la creación de nodos
    // (no podemos reducir poolSize < 3, pero podemos pausar renders no críticos)
    console.log('🗑️ GC pressure: deferring non-critical renders');
    // Aquí podríamos implementar una cola de prioridad para renders
  }
  
  // Método para que el pool consulte el throttle actual
  getCurrentConfig() {
    return {
      throttleMs: this.currentThrottleMs,
      poolSize: this.currentPoolSize,
      overscan: this.currentOverscan
    };
  }
}
```

## ⚠️ NOTAS CRÍTICAS PARA IMPLEMENTACIÓN

1. **Safari 14 NO tiene `performance.memory`**: Todas las métricas de RAM son heurísticas. Validar empíricamente con Xcode Instruments.

2. **`requestIdleCallback` NO está en Safari 14**: Usar fallback a `setTimeout(fn, 100)` con prioridad baja.

3. **Object URLs para blobs**: Siempre llamar `URL.revokeObjectURL()` tras usar imágenes de IndexedDB, o memory leak garantizado en 2GB RAM.

4. **Transform vs Top**: Usar SIEMPRE `transform: translateY()` para reposicionar nodos del pool. `top/left` disparan reflow.

5. **Contain CSS**: `contain: strict` en el viewport y `contain: content` en los nodos del pool es OBLIGATORIO para aislar repaints.

6. **Testing en hardware real**: Chrome DevTools throttling NO replica el thermal throttling de iPad A10. Probar con dispositivo físico bajo carga solar (sí, en serio).
