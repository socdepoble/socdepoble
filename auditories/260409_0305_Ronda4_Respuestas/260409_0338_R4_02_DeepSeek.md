## 📐 FRENTE 1: VIRTUAL SCROLLING MATEMÁTICO (VANILLA JS)

### 1.1 Pre‑cálculo de alturas (Node.js build pipeline)

No podemos medir alturas en cliente sin renderizar los fragmentos, porque eso causaría layout thrashing. Solución: **pre‑calcular las alturas de cada fragmento durante el build** usando Puppeteer (headless) y guardarlas en `manifest.json`.

```javascript
// build-heightmap.js
const puppeteer = require('puppeteer');
const fs = require('fs');

(async () => {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();
  await page.setViewport({ width: 1024, height: 768, deviceScaleFactor: 1 });
  
  const manifest = JSON.parse(fs.readFileSync('public/fragments/manifest.json'));
  const heightMap = [];
  
  for (const frag of manifest) {
    const html = fs.readFileSync(`public/fragments/${frag.id}.html`, 'utf8');
    await page.setContent(html);
    const height = await page.evaluate(() => {
      const body = document.body;
      const style = window.getComputedStyle(body);
      return body.scrollHeight + parseInt(style.marginTop) + parseInt(style.marginBottom);
    });
    heightMap.push({ id: frag.id, height, order: frag.order });
  }
  
  manifest.heightMap = heightMap;
  fs.writeFileSync('public/fragments/manifest.json', JSON.stringify(manifest, null, 2));
  await browser.close();
})();
```

### 1.2 Virtual Scroller con 3 nodos reciclados

El núcleo: un array de alturas (`heightMap`), un contenedor `position: relative` con una altura total `sum(heights)`, y **3 divs absolutos** que se reposicionan con `transform: translateY` según el índice visible.

```javascript
// virtual-scroller.js
class VirtualScroller {
  constructor(container, manifest, loadFragmentCallback) {
    this.container = container;        // elemento DOM contenedor (position: relative)
    this.heightMap = manifest.heightMap;
    this.totalHeight = this.heightMap.reduce((acc, h) => acc + h.height, 0);
    this.loadFragment = loadFragmentCallback; // async (index) => HTML string
    this.cachedNodes = new Map();       // guarda los nodos ya creados (máximo 3 + buffer)
    this.activeIndices = new Set();     // índices actualmente visibles (máximo 3)
    this.nodePool = [];                 // divs reutilizables (siempre 3)
    
    this._setupDOM();
    this._attachScrollHandler();
  }
  
  _setupDOM() {
    this.container.style.position = 'relative';
    this.container.style.height = `${this.totalHeight}px`;
    this.container.style.overflow = 'auto';
    this.container.innerHTML = '';      // vaciar contenido previo
    for (let i = 0; i < 3; i++) {
      const div = document.createElement('div');
      div.style.position = 'absolute';
      div.style.top = '0';
      div.style.left = '0';
      div.style.width = '100%';
      div.style.willChange = 'transform';
      div.style.contain = 'layout style'; // aislar reflow
      this.container.appendChild(div);
      this.nodePool.push(div);
    }
  }
  
  _attachScrollHandler() {
    let ticking = false;
    this.container.addEventListener('scroll', () => {
      if (!ticking) {
        requestAnimationFrame(() => {
          this._updateVisibleNodes();
          ticking = false;
        });
        ticking = true;
      }
    });
    this._updateVisibleNodes(); // inicial
  }
  
  _updateVisibleNodes() {
    const scrollTop = this.container.scrollTop;
    const viewportHeight = this.container.clientHeight;
    const startPos = scrollTop;
    const endPos = scrollTop + viewportHeight;
    
    // Encontrar índices visibles mediante búsqueda binaria (heightMap acumulativo)
    const startIndex = this._findIndexByOffset(startPos);
    const endIndex = this._findIndexByOffset(endPos);
    
    const newIndices = new Set();
    for (let i = startIndex; i <= endIndex && i < this.heightMap.length; i++) {
      newIndices.add(i);
    }
    
    // Añadir un buffer de 1 arriba y abajo para evitar parpadeos
    if (startIndex > 0) newIndices.add(startIndex - 1);
    if (endIndex + 1 < this.heightMap.length) newIndices.add(endIndex + 1);
    
    // Calcular qué índices ya no están visibles
    const toRemove = [...this.activeIndices].filter(i => !newIndices.has(i));
    const toAdd = [...newIndices].filter(i => !this.activeIndices.has(i));
    
    // Reciclar nodos: reutilizar los que se liberan
    toRemove.forEach(idx => {
      const node = this.cachedNodes.get(idx);
      if (node && node.parentNode) {
        // No lo eliminamos del DOM, solo lo escondemos (pero lo pondremos en otro lugar)
        // En realidad lo reutilizaremos para los nuevos índices
        this.cachedNodes.delete(idx);
        this.nodePool.push(node);
      }
    });
    
    toAdd.forEach(idx => {
      let node = this.nodePool.pop();
      if (!node) {
        // Fallback: crear nuevo (nunca debería pasar porque siempre tenemos 3)
        node = document.createElement('div');
        node.style.position = 'absolute';
        node.style.left = '0';
        node.style.width = '100%';
        node.style.willChange = 'transform';
        this.container.appendChild(node);
      }
      // Calcular offsetY acumulado hasta este índice
      const offsetY = this._getOffsetAtIndex(idx);
      node.style.transform = `translateY(${offsetY}px)`;
      
      // Cargar contenido si no está ya cacheado
      if (!node.hasAttribute('data-loaded')) {
        this.loadFragment(idx).then(html => {
          node.innerHTML = html;
          node.setAttribute('data-loaded', 'true');
          // Ejecutar highlight solo dentro de este nodo
          if (window.hljs) {
            node.querySelectorAll('pre code').forEach(block => hljs.highlightElement(block));
          }
        }).catch(err => console.error('Error loading fragment', idx, err));
      }
      this.cachedNodes.set(idx, node);
    });
    
    this.activeIndices = newIndices;
  }
  
  _findIndexByOffset(offset) {
    // búsqueda binaria sobre alturas acumuladas
    let low = 0, high = this.heightMap.length - 1;
    while (low <= high) {
      const mid = Math.floor((low + high) / 2);
      const accStart = this._getOffsetAtIndex(mid);
      const accEnd = accStart + this.heightMap[mid].height;
      if (offset >= accStart && offset < accEnd) return mid;
      if (offset < accStart) high = mid - 1;
      else low = mid + 1;
    }
    return Math.min(this.heightMap.length - 1, Math.max(0, low));
  }
  
  _getOffsetAtIndex(idx) {
    let sum = 0;
    for (let i = 0; i < idx; i++) sum += this.heightMap[i].height;
    return sum;
  }
}
```

### 1.3 Integración con el Queue Manager (Ronda 3)

El callback `loadFragment` debe usar el **Queue Manager** con concurrencia limitada (2 peticiones simultáneas) y caché IDB/Cache.

```javascript
async function loadFragmentByIndex(idx) {
  const fragId = manifest.heightMap[idx].id;
  // Primero intentar desde IndexedDB (fragment cache)
  let html = await getFragmentFromIDB(fragId);
  if (!html) {
    html = await enqueueFragmentLoad(fragId); // usa fetch con cola
    await setFragmentToIDB(fragId, html);
  }
  return html;
}
```

**Fallo catastrófico previsto:** Si la altura de un fragmento cambia dinámicamente (por ejemplo, carga de imágenes diferidas o cambio de fuente por accesibilidad), el `heightMap` precalculado será incorrecto. Solución: después de cargar cada fragmento, volver a medir su altura real y **actualizar el heightMap en vivo** (reajustando `translateY` de todos los nodos siguientes). Eso requiere una pasada de actualización costosa pero inevitable.

## 📊 FRENTE 2: BENCHMARKING Y THERMAL THROTTLE

### 2.1 Detección de caída de FPS con `requestAnimationFrame`

Safari 14 no soporta `PerformanceObserver` para `layout-shift` ni `longtask`, pero podemos medir FPS con un bucle de RAF.

```javascript
// fps-monitor.js
class ThermalMonitor {
  constructor(thresholdFPS = 30, sampleInterval = 2000) {
    this.thresholdFPS = thresholdFPS;
    this.sampleInterval = sampleInterval; // ms
    this.frameTimes = [];
    this.lastFrameTime = performance.now();
    this.isThermalThrottling = false;
    this._rafId = null;
    this._listeners = [];
  }
  
  start() {
    this._rafId = requestAnimationFrame(this._measure.bind(this));
    setInterval(() => this._checkThermal(), this.sampleInterval);
  }
  
  _measure(now) {
    const delta = now - this.lastFrameTime;
    if (delta > 0) {
      this.frameTimes.push(delta);
      if (this.frameTimes.length > 60) this.frameTimes.shift(); // guardar últimas 60 muestras
    }
    this.lastFrameTime = now;
    this._rafId = requestAnimationFrame(this._measure.bind(this));
  }
  
  _checkThermal() {
    if (this.frameTimes.length < 30) return;
    const avgDelta = this.frameTimes.reduce((a,b) => a+b, 0) / this.frameTimes.length;
    const fps = 1000 / avgDelta;
    const wasThrottling = this.isThermalThrottling;
    this.isThermalThrottling = (fps < this.thresholdFPS);
    
    if (this.isThermalThrottling !== wasThrottling) {
      this._dispatchEvent(this.isThermalThrottling ? 'thermal:start' : 'thermal:end');
    }
  }
  
  _dispatchEvent(type) {
    this._listeners.forEach(fn => fn(type));
  }
  
  on(callback) {
    this._listeners.push(callback);
  }
}
```

### 2.2 Detección de pausas del Garbage Collector

El GC bloquea el hilo principal; podemos detectarlo midiendo **diferencias anómalas entre frames** (delta > 100ms).

```javascript
// Dentro de _measure(now)
if (delta > 100) {
  // Posible GC o bloqueo del hilo
  console.warn(`Long frame detected: ${delta}ms`);
  this._dispatchEvent('longframe', delta);
}
```

### 2.3 Activación del modo `thermal_throttle`

Cuando se detecta throttling, el sistema debe:
- Reducir la calidad de imágenes (usando `srcset` con tamaños más pequeños).
- Limitar la precarga de fragmentos a solo 1 concurrencia.
- Desactivar animaciones CSS y efectos de sombra.
- Reducir el `rootMargin` del IntersectionObserver a 100px.

```javascript
thermalMonitor.on((event) => {
  if (event === 'thermal:start') {
    document.body.classList.add('thermal-throttle');
    // Reducir concurrencia de fetch
    window.MAX_CONCURRENT_LOADS = 1;
    // Desactivar precarga agresiva
    if (window.observer) window.observer.rootMargin = '100px';
  } else if (event === 'thermal:end') {
    document.body.classList.remove('thermal-throttle');
    window.MAX_CONCURRENT_LOADS = 2;
    if (window.observer) window.observer.rootMargin = '300px';
  }
});
```

### 2.4 Medición de Layout Thrashing

Podemos usar `PerformanceObserver` (si está disponible) para `layout-shift`, pero en Safari 14 no. Alternativa: medir el tiempo entre `scroll` y la siguiente pintura.

```javascript
let scrollStart = 0;
window.addEventListener('scroll', () => {
  scrollStart = performance.now();
  requestAnimationFrame(() => {
    const paintDelay = performance.now() - scrollStart;
    if (paintDelay > 50) {
      console.warn(`Layout thrashing detected: ${paintDelay}ms`);
      // forzar reducción de nodos activos
      if (window.virtualScroller && window.virtualScroller.activeIndices.size > 2) {
        window.virtualScroller.activeIndices = new Set([...window.virtualScroller.activeIndices].slice(0,2));
      }
    }
  });
});
```
