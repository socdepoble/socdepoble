## 🛡️ FRENTE 1: VIRTUAL SCROLLING MATEMÁTICO ABSOLUTO

### Arquitectura: "El Trío de la Muerte" (Object Pooling de 3 Nodos)

El objetivo es mantener exactamente **3 nodos DOM** en memoria, sin importar el tamaño del corpus (200K líneas). Estos nodos son contenedores absolutos que se reciclan en tiempo real mediante `transform: translateY()`.

```javascript
/**
 * GENOTIPO SINTÉTICO: VIRTUAL SCROLLER ABSOLUTO
 * Hardware target: iPad A10 2GB RAM / Safari 14
 * Nodos DOM máximos: 3 (viewport + buffer previo + buffer posterior)
 */

class GenotypeVirtualScroller {
  constructor(containerId, heightMap, renderFn) {
    this.container = document.getElementById(containerId);
    this.heightMap = heightMap; // Array de alturas precalculadas [200, 4000, 800...]
    this.renderFn = renderFn;   // Función que genera contenido HTML
    
    // Pool de exactamente 3 nodos
    this.pool = [];
    this.viewportHeight = window.innerHeight;
    this.scrollTop = 0;
    this.totalHeight = this.heightMap.reduce((a, b) => a + b, 0);
    
    // Métricas para sincronización RAF
    this.lastScrollTime = 0;
    this.scrollVelocity = 0;
    this.pendingUpdate = null;
    
    this.init();
  }

  init() {
    // Crear viewport fantasma kilométrico
    this.ghost = document.createElement('div');
    this.ghost.style.height = `${this.totalHeight}px`;
    this.ghost.style.position = 'relative';
    this.container.appendChild(this.ghost);

    // Crear pool de 3 nodos absolutos
    for (let i = 0; i < 3; i++) {
      const node = document.createElement('div');
      node.style.position = 'absolute';
      node.style.left = '0';
      node.style.width = '100%';
      node.style.willChange = 'transform'; // GPU acceleration
      node.style.contain = 'strict'; // CSS containment crítico
      node.style.contentVisibility = 'auto'; // Safari 14+ soporta esto
      this.pool.push(node);
      this.ghost.appendChild(node);
    }

    // Throttled scroll listener (16ms = ~60fps max input)
    this.container.addEventListener('scroll', this.onScroll.bind(this), { passive: true });
    
    // Inicializar posiciones
    this.updateVisibleNodes();
  }

  /**
   * Búsqueda binaria O(log n) para encontrar índice en mapa de alturas acumuladas
   * Crítico para arrays de 200K elementos
   */
  binarySearchCumulativeHeight(targetScroll) {
    let left = 0, right = this.heightMap.length;
    let cumulative = 0;
    
    // Pre-calcular array de alturas acumuladas si no existe
    if (!this.cumulativeHeights) {
      this.cumulativeHeights = new Float64Array(this.heightMap.length);
      let sum = 0;
      for (let i = 0; i < this.heightMap.length; i++) {
        this.cumulativeHeights[i] = sum;
        sum += this.heightMap[i];
      }
    }

    // Búsqueda binaria en Float64Array (muy rápido en JSC de Safari)
    while (left < right) {
      const mid = (left + right) >>> 1;
      if (this.cumulativeHeights[mid] < targetScroll) {
        left = mid + 1;
      } else {
        right = mid;
      }
    }
    return Math.max(0, left - 1);
  }

  onScroll(e) {
    const now = performance.now();
    this.scrollTop = this.container.scrollTop;
    
    // Calcular velocidad para predicción
    this.scrollVelocity = (this.scrollTop - (this.lastScrollTop || 0)) / (now - this.lastScrollTime);
    this.lastScrollTime = now;
    this.lastScrollTop = this.scrollTop;

    // Cancelar RAF pendiente si existe (evita acumulación)
    if (this.pendingUpdate) {
      cancelAnimationFrame(this.pendingUpdate);
    }

    // Throttle: solo actualizar si pasaron > 8ms (120fps cap) o velocidad es extrema
    if (now - this.lastRenderTime > 8 || Math.abs(this.scrollVelocity) > 50) {
      this.pendingUpdate = requestAnimationFrame(() => {
        this.updateVisibleNodes();
        this.lastRenderTime = performance.now();
        this.pendingUpdate = null;
      });
    }
  }

  updateVisibleNodes() {
    const viewportTop = this.scrollTop;
    const viewportBottom = viewportTop + this.viewportHeight;
    
    // Encontrar índice del elemento visible central
    const centerIndex = this.binarySearchCumulativeHeight(viewportTop + this.viewportHeight / 2);
    
    // Determinar qué índices deben ocupar los 3 nodos
    // Nodo 0: anterior (si existe), Nodo 1: actual, Nodo 2: posterior
    const indices = [
      Math.max(0, centerIndex - 1),
      centerIndex,
      Math.min(this.heightMap.length - 1, centerIndex + 1)
    ];

    // Reciclar nodos
    this.pool.forEach((node, i) => {
      const idx = indices[i];
      const yPos = this.cumulativeHeights[idx];
      const height = this.heightMap[idx];
      
      // Solo actualizar si cambió el índice o posición
      if (node.dataset.index !== String(idx)) {
        node.dataset.index = idx;
        node.style.transform = `translateY(${yPos}px)`;
        node.style.height = `${height}px`;
        
        // Renderizado lazy con IntersectionObserver ya no es necesario
        // Ahora es síncrono pero con contenido pre-cargado desde IDB
        this.renderFn(node, idx);
      }
    });
  }

  /**
   * Método para actualizar altura dinámica (si el contenido cambia de tamaño)
   * Recalcula cumulativeHeights parcialmente
   */
  updateHeightAtIndex(index, newHeight) {
    const delta = newHeight - this.heightMap[index];
    this.heightMap[index] = newHeight;
    
    // Actualizar cumulative desde index+1 hasta el final
    for (let i = index + 1; i < this.cumulativeHeights.length; i++) {
      this.cumulativeHeights[i] += delta;
    }
    this.totalHeight += delta;
    this.ghost.style.height = `${this.totalHeight}px`;
  }
}
```

```javascript
// scripts/genotype-builder.js
const fs = require('fs');
const path = require('path');
const { JSDOM } = require('jsdom');

const MONOLOTH_PATH = './src/tomo-completo.html';
const OUTPUT_DIR = './dist/chunks';
const CHUNK_SIZE = 50; // artículos por chunk

async function genotypeSurgery() {
  const html = fs.readFileSync(MONOLOTH_PATH, 'utf-8');
  const dom = new JSDOM(html);
  const document = dom.window.document;
  
  const articles = Array.from(document.querySelectorAll('article[data-genotype-id]'));
  const heightMap = new Float64Array(articles.length);
  const chunkManifest = [];

  // Simular renderizado para calcular alturas
  const virtualContainer = document.createElement('div');
  virtualContainer.style.cssText = 'position:absolute;visibility:hidden;width:768px;'; // iPad width
  document.body.appendChild(virtualContainer);

  for (let i = 0; i < articles.length; i += CHUNK_SIZE) {
    const chunk = articles.slice(i, i + CHUNK_SIZE);
    const chunkData = [];
    
    chunk.forEach((article, offset) => {
      const globalIndex = i + offset;
      
      // Clonar y medir
      const clone = article.cloneNode(true);
      virtualContainer.appendChild(clone);
      const height = clone.getBoundingClientRect().height;
      heightMap[globalIndex] = Math.ceil(height);
      
      // Extraer contenido esencial (sin el HTML completo, solo datos)
      chunkData.push({
        id: article.dataset.genotypeId,
        height: heightMap[globalIndex],
        html: clone.innerHTML, // Contenido limpio
        metadata: {
          title: article.querySelector('h2')?.textContent || '',
          category: article.dataset.category
        }
      });
      
      clone.remove();
    });

    // Guardar chunk como JSON comprimido
    const chunkPath = path.join(OUTPUT_DIR, `chunk-${Math.floor(i/CHUNK_SIZE)}.json`);
    fs.writeFileSync(chunkPath, JSON.stringify(chunkData));
    chunkManifest.push({
      index: Math.floor(i/CHUNK_SIZE),
      startArticle: i,
      endArticle: Math.min(i + CHUNK_SIZE, articles.length) - 1,
      path: `chunks/chunk-${Math.floor(i/CHUNK_SIZE)}.json`,
      byteSize: fs.statSync(chunkPath).size
    });
  }

  virtualContainer.remove();

  // Generar manifest maestro
  const manifest = {
    version: 'genotype-4.0',
    totalArticles: articles.length,
    totalHeight: Array.from(heightMap).reduce((a,b) => a+b, 0),
    heightMap: Array.from(heightMap), // Será cargado como Float64Array en cliente
    chunks: chunkManifest,
    generatedAt: new Date().toISOString()
  };

  fs.writeFileSync('./dist/genotype-manifest.json', JSON.stringify(manifest));
  console.log(`✅ Genotipo fragmentado: ${articles.length} artículos en ${chunkManifest.length} chunks`);
}

genotypeSurgery();
```

### PerformanceObserver + RAF Delta para Safari 14

```javascript
/**
 * THERMAL DEATH WATCHDOG
 * Monitorea señales de estrés térmico en iPad A10
 */

class ThermalWatchdog {
  constructor() {
    this.samples = [];
    this.maxSamples = 60; // 1 segundo de historial a 60fps
    this.thermalState = 'normal'; // normal | warning | critical
    this.gcHints = [];
    
    this.fpsHistory = new Float64Array(60);
    this.fpsIndex = 0;
    
    this.init();
  }

  init() {
    // 1. FPS Monitoring via RAF deltas
    this.trackFPS();
    
    // 2. Layout Thrashing Detection
    this.detectLayoutThrashing();
    
    // 3. Memory Pressure (indirecta en Safari)
    this.monitorMemoryPressure();
    
    // 4. Long Task Detection (PerformanceObserver)
    if ('PerformanceObserver' in window) {
      this.observeLongTasks();
    }
  }

  trackFPS() {
    let lastTime = performance.now();
    let frames = 0;
    let lastFpsTime = lastTime;

    const measure = (currentTime) => {
      frames++;
      
      // Calcular FPS cada 1000ms
      if (currentTime - lastFpsTime >= 1000) {
        const fps = frames;
        frames = 0;
        lastFpsTime = currentTime;
        
        // Guardar en buffer circular
        this.fpsHistory[this.fpsIndex] = fps;
        this.fpsIndex = (this.fpsIndex + 1) % 60;
        
        // Detectar caída térmica
        this.analyzeThermalState(fps);
      }
      
      requestAnimationFrame(measure);
    };
    
    requestAnimationFrame(measure);
  }

  analyzeThermalState(currentFps) {
    const avgFps = this.fpsHistory.reduce((a,b) => a+b, 0) / 60;
    const recentAvg = this.getRecentAverage(10); // últimos 10 segundos
    
    // Umbrales para iPad A10 (Safari 14 típicamente hace 60fps en idle)
    if (currentFps < 20 || recentAvg < 30) {
      this.thermalState = 'critical';
      this.triggerThermalThrottle();
    } else if (currentFps < 30 || recentAvg < 45) {
      this.thermalState = 'warning';
      this.triggerWarning();
    } else {
      this.thermalState = 'normal';
    }
  }

  getRecentAverage(seconds) {
    let sum = 0;
    for (let i = 0; i < seconds; i++) {
      const idx = (this.fpsIndex - 1 - i + 60) % 60;
      sum += this.fpsHistory[idx] || 60;
    }
    return sum / seconds;
  }

  /**
   * Detección de Layout Thrashing: lectura/escritura alternada al DOM
   */
  detectLayoutThrashing() {
    const originalOffsetHeight = Object.getOwnPropertyDescriptor(
      HTMLElement.prototype, 'offsetHeight'
    );
    
    let readCount = 0;
    let writeCount = 0;
    let lastPhase = null; // 'read' | 'write'

    // Instrumentar getters de layout
    ['offsetHeight', 'offsetWidth', 'clientHeight', 'scrollHeight', 'getBoundingClientRect'].forEach(prop => {
      const original = HTMLElement.prototype[prop] || 
                      (prop === 'getBoundingClientRect' ? Element.prototype[prop] : null);
      
      if (!original) return;
      
      const instrumented = function() {
        if (lastPhase === 'write') {
          // THRASHING DETECTED: lectura después de escritura
          console.warn('🚨 LAYOUT THRASHING:', prop, 'llamado después de modificación DOM');
          this.thrashingCount = (this.thrashingCount || 0) + 1;
        }
        lastPhase = 'read';
        readCount++;
        return original.apply(this, arguments);
      };
      
      if (prop === 'getBoundingClientRect') {
        Element.prototype[prop] = instrumented;
      } else {
        Object.defineProperty(HTMLElement.prototype, prop, {
          get: instrumented
        });
      }
    });

    // Monitorear métodos de escritura
    const writeMethods = ['appendChild', 'insertBefore', 'removeChild', 'innerHTML', 'setAttribute'];
    writeMethods.forEach(method => {
      const original = HTMLElement.prototype[method];
      HTMLElement.prototype[method] = function(...args) {
        lastPhase = 'write';
        writeCount++;
        return original.apply(this, args);
      };
    });

    // Reporte periódico
    setInterval(() => {
      if (readCount > 0 || writeCount > 0) {
        const thrashRatio = (this.thrashingCount || 0) / Math.max(readCount, 1);
        if (thrashRatio > 0.3) {
          this.thermalState = 'critical';
          this.triggerThermalThrottle();
        }
        readCount = writeCount = 0;
        this.thrashingCount = 0;
      }
    }, 1000);
  }

  /**
   * Memory Pressure via Performance.memory (Chrome) o heurísticas (Safari)
   */
  monitorMemoryPressure() {
    if (performance.memory) {
      // Chrome/Edge
      setInterval(() => {
        const used = performance.memory.usedJSHeapSize;
        const total = performance.memory.totalJSHeapSize;
        const limit = performance.memory.jsHeapSizeLimit;
        
        if (used / limit > 0.8) {
          this.triggerThermalThrottle();
        }
      }, 2000);
    } else {
      // Safari: usar heurísticas de GC timing
      this.gcTimingHeuristic();
    }
  }

  gcTimingHeuristic() {
    // En Safari, los pausas largas del main thread sugieren GC
    let lastTime = performance.now();
    
    const checkGaps = () => {
      const now = performance.now();
      const gap = now - lastTime;
      
      // Si el gap es > 50ms, probablemente hubo GC o thermal throttle del OS
      if (gap > 50 && gap < 1000) {
        this.gcHints.push({ time: now, duration: gap });
        if (this.gcHints.length > 10) this.gcHints.shift();
        
        // Si hay 3+ pausas en 5 segundos, estamos en problemas
        const recent = this.gcHints.filter(h => now - h.time < 5000);
        if (recent.length >= 3) {
          this.triggerThermalThrottle();
        }
      }
      
      lastTime = now;
      requestAnimationFrame(checkGaps);
    };
    
    requestAnimationFrame(checkGaps);
  }

  observeLongTasks() {
    if (!('PerformanceObserver' in window)) return;
    
    try {
      const observer = new PerformanceObserver((list) => {
        for (const entry of list.getEntries()) {
          // Long Task > 50ms bloquea el main thread
          if (entry.duration > 50) {
            console.warn('⏱️ LONG TASK:', entry.duration.toFixed(2), 'ms');
            if (entry.duration > 100) {
              this.thermalState = 'warning';
            }
            if (entry.duration > 200) {
              this.thermalState = 'critical';
              this.triggerThermalThrottle();
            }
          }
        }
      });
      
      observer.observe({ entryTypes: ['longtask'] });
    } catch (e) {
      // Safari 14 puede no soportar longtask
    }
  }

  triggerThermalThrottle() {
    if (this.throttling) return;
    this.throttling = true;
    
    console.warn('🔥 THERMAL THROTTLE ACTIVADO');
    
    // Emitir evento global para que el VirtualScroller se adapte
    window.dispatchEvent(new CustomEvent('thermal-throttle', {
      detail: { state: this.thermalState }
    }));
    
    // Reducir calidad temporalmente
    this.emergencyMeasures();
    
    // Recuperación gradual
    setTimeout(() => {
      this.throttling = false;
      this.thermalState = 'normal';
    }, 3000);
  }

  emergencyMeasures() {
    // 1. Pausar carga de chunks desde IDB
    window.dispatchEvent(new Event('pause-chunk-loading'));
    
    // 2. Reducir buffer del VirtualScroller a 1 nodo
    window.dispatchEvent(new CustomEvent('reduce-buffer', { detail: { nodes: 1 } }));
    
    // 3. Forzar GC sugerido (eliminar referencias)
    if (window.gc) window.gc(); // Solo en Chrome con flag
    
    // 4. Reducir resolución de imágenes si las hay
    document.body.classList.add('thermal-mode');
  }

  getReport() {
    return {
      thermalState: this.thermalState,
      avgFps: this.getRecentAverage(60),
      recentFps: Array.from(this.fpsHistory),
      gcHints: this.gcHints.length,
      timestamp: performance.now()
    };
  }
}
```

### Integración Híbrida IDB + Cache API con Backpressure

```javascript
/**
 * HYBRID STORAGE MANAGER con control de presión térmica
 */

class HybridChunkManager {
  constructor() {
    this.idb = null;
    this.cache = null;
    this.chunkCache = new Map(); // LRU en memoria (máx 3 chunks)
    this.maxMemoryChunks = 3;
    this.loadingQueue = [];
    this.isThrottled = false;
    
    this.init();
  }

  async init() {
    // Abrir IndexedDB
    this.idb = await openDB('genotype-chunks', 1, {
      upgrade(db) {
        db.createObjectStore('chunks', { keyPath: 'id' });
      }
    });
    
    // Abrir Cache API
    this.cache = await caches.open('genotype-v4');
    
    // Escuchar eventos térmicos
    window.addEventListener('thermal-throttle', () => {
      this.isThrottled = true;
      this.loadingQueue = []; // Cancelar cola
    });
    
    window.addEventListener('pause-chunk-loading', () => {
      this.loadingPaused = true;
    });
  }

  async getChunk(chunkIndex) {
    // 1. Verificar memoria (más rápido)
    if (this.chunkCache.has(chunkIndex)) {
      return this.chunkCache.get(chunkIndex);
    }

    // 2. Si estamos throttled, no cargar nuevos chunks
    if (this.isThrottled && this.chunkCache.size >= 1) {
      return null; // Devolver null, el renderer mostrará placeholder
    }

    // 3. Intentar IDB
    try {
      const fromIdb = await this.idb.get('chunks', chunkIndex);
      if (fromIdb) {
        this.storeInMemory(chunkIndex, fromIdb);
        return fromIdb;
      }
    } catch (e) {
      // IDB puede estar bloqueado
    }

    // 4. Fetch de red con Cache API fallback
    if (!this.loadingPaused) {
      return await this.fetchWithCache(chunkIndex);
    }
    
    return null;
  }

  async fetchWithCache(chunkIndex) {
    const url = `/chunks/chunk-${chunkIndex}.json`;
    
    try {
      // Intentar Cache API primero (más rápido que IDB para archivos)
      const cached = await this.cache.match(url);
      if (cached) {
        const data = await cached.json();
        this.storeInMemory(chunkIndex, data);
        this.storeInIDB(chunkIndex, data); // Background
        return data;
      }
      
      // Fetch de red
      const response = await fetch(url);
      const data = await response.json();
      
      // Guardar en ambos stores (background)
      this.cache.put(url, response.clone());
      this.storeInIDB(chunkIndex, data);
      this.storeInMemory(chunkIndex, data);
      
      return data;
    } catch (e) {
      return null;
    }
  }

  storeInMemory(index, data) {
    // LRU eviction si excedemos límite
    if (this.chunkCache.size >= this.maxMemoryChunks) {
      const firstKey = this.chunkCache.keys().next().value;
      this.chunkCache.delete(firstKey);
    }
    this.chunkCache.set(index, data);
  }

  async storeInIDB(index, data) {
    try {
      await this.idb.put('chunks', { id: index, data, storedAt: Date.now() });
    } catch (e) {
      // IDB puede estar lleno o bloqueado
    }
  }
}
```

## ⚔️ FISURAS DETECTADAS Y CONTRAMEDIDAS

| Fisura | Riesgo | Contramedida en Frente 2 |
|--------|--------|-------------------------|
| Búsqueda binaria O(log n) puede ser lenta si el mapa no cabe en caché L1 | 200K elementos × 8 bytes = 1.6MB, excede L1 | Usar `Float64Array` que optimiza bien en JSC, dividir en segmentos de 1000 |
| RAF acumulativo durante swipe agresivo | Cola de RAFs pendientes crece exponencialmente | Cancelar RAF anterior con `cancelAnimationFrame` |
| IDB transactions bloquean main thread | Safari 14 tiene IDB lento | Cache API como primera línea, IDB como backup |
| Garbage Collector de Safari es conservador | OOM silencioso antes de GC agresivo | `null` explícito a referencias, pools reutilizados |
