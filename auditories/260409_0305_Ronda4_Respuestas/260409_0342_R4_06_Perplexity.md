## Frente 1: Virtual Scrolling Matemático

### Arquitectura Final: Viewport Fantasma + Object Pooling
**Concepto base:** Un `<div id="viewport">` de altura fija (ej: 100vh) + contenedor fantasma `#scrollContainer` de altura total computada (suma alturas chunks). Solo 3 `div`s físicos reutilizados via `transform: translateY(px)` precisa. Scroll sincronizado via `requestAnimationFrame` + mapa acumulativo de alturas.

```js
const fs = require('fs');
const cheerio = require('cheerio'); // npm i cheerio
const html = fs.readFileSync('llibre-sencer.html', 'utf8');
const $ = cheerio.load(html);
const articles = $('article, section').map((i, el) => $(el).html()).get();
const chunkSize = 5; // 5000 líneas ≈ 5 articles grandes
const chunks = [];
for(let i=0; i<articles.length; i+=chunkSize) {
  chunks.push({id: i/chunkSize, html: articles.slice(i,i+chunkSize).join('')});
}
fs.writeFileSync('chunks.json', JSON.stringify(chunks));
chunks.forEach((c,i) => 
  fs.writeFileSync(`chunks/${c.id}.html`, `<div data-chunk="${c.id}">${c.html}</div>`)
);
```

```js
// virtual-core.js - Virtual Scrolling Absoluto
class VirtualScroll {
  constructor() {
    this.chunkHeightMap = []; // [200, 4000, 800, ...] alturas reales
    this.totalHeight = 0;
    this.pool = []; // 3 nodos reutilizables
    this.visibleChunks = new Set(); // {0,1,2} chunks en viewport
    this.scrollTop = 0;
    
    this.initPool();
    this.loadChunkHeights(); // Desde chunks.json cacheado
    this.totalHeight = this.chunkHeightMap.reduce((a,b)=>a+b, 0);
    document.getElementById('scrollContainer').style.height = `${this.totalHeight}px`;
    
    this.viewport = document.getElementById('viewport');
    this.bindScroll();
    this.renderLoop();
  }
  
  initPool() {
    const container = document.getElementById('scrollContainer');
    for(let i=0; i<3; i++) {
      const node = document.createElement('div');
      node.className = 'pool-node';
      container.appendChild(node);
      this.pool.push({node, chunkId: -1, top: 0});
    }
  }
  
  // Mapa acumulativo: chunk 0 @ 0px, chunk 1 @ 1200px, etc.
  getChunkAtPosition(pos) {
    let sum = 0, chunkId = 0;
    for(; chunkId<this.chunkHeightMap.length; chunkId++) {
      if(pos < sum + this.chunkHeightMap[chunkId]) return chunkId;
      sum += this.chunkHeightMap[chunkId];
    }
    return chunkId-1;
  }
  
  bindScroll() {
    let ticking = false;
    this.viewport.addEventListener('scroll', () => {
      if(!ticking) {
        requestAnimationFrame(() => {
          this.scrollTop = this.viewport.scrollTop;
          this.updateVisibleChunks();
          ticking = false;
        });
        ticking = true;
      }
    });
  }
  
  updateVisibleChunks() {
    const startChunk = this.getChunkAtPosition(this.scrollTop);
    const endChunk = this.getChunkAtPosition(this.scrollTop + window.innerHeight * 2); // Preload
    
    // Prune chunks fuera de rango
    for(let id of this.visibleChunks) {
      if(id < startChunk || id > endChunk) {
        this.releaseChunk(id);
      }
    }
    
    // Allocate chunks en rango
    for(let id=startChunk; id<=endChunk; id++) {
      if(!this.visibleChunks.has(id)) this.acquireChunk(id);
    }
  }
  
  async acquireChunk(chunkId) {
    const node = this.pool.find(p => p.chunkId === -1);
    if(!node) return; // Pool lleno
    
    const html = await this.fetchChunk(chunkId); // Cache/SW
    node.chunkId = chunkId;
    node.node.innerHTML = html;
    node.node.style.height = `${this.chunkHeightMap[chunkId]}px`;
    
    const top = this.chunkHeightMap.slice(0, chunkId).reduce((a,b)=>a+b, 0);
    node.top = top;
    node.node.style.transform = `translateY(${top}px)`;
    
    this.visibleChunks.add(chunkId);
  }
  
  releaseChunk(chunkId) {
    const node = this.pool.find(p => p.chunkId === chunkId);
    node.node.innerHTML = '';
    node.chunkId = -1;
    this.visibleChunks.delete(chunkId);
  }
  
  renderLoop() {
    // RAF loop mantiene 60FPS sync
    requestAnimationFrame(() => this.renderLoop());
  }
  
  async fetchChunk(id) {
    // Híbrido Cache API -> IDB -> fallback
    const cache = await caches.match(`chunk-${id}.html`);
    if(cache) return cache.text();
    // IDB fallback aquí...
    return '';
  }
}

new VirtualScroll();
```

**Sincronización Matemática:** `getChunkAtPosition()` es O(n) pero n=40 chunks → negligible. Alturas precomputadas en `chunkHeightMap`. Pool limitado a 3 nodos → RAM constante 60MB max.

```js
// performance-monitor.js - Monitoreo en vivo
class PerformanceMonitor {
  constructor() {
    this.fpsHistory = [];
    this.layoutCount = 0;
    this.gcPauseHistory = [];
    this.thermalState = 'normal';
    
    this.startFPSMeter();
    this.observeLayoutThrashing();
    this.watchThermalThrottle();
  }
  
  startFPSMeter() {
    let lastTime = performance.now();
    let frameCount = 0;
    
    const measureFPS = (now) => {
      frameCount++;
      if(now - lastTime >= 1000) {
        const fps = frameCount * 1000 / (now - lastTime);
        this.fpsHistory.push(fps);
        if(this.fpsHistory.length > 60) this.fpsHistory.shift();
        
        // Thermal throttle si FPS < 25 por 3s
        if(fps < 25 && this.fpsHistory.slice(-10).every(f=>f<25)) {
          this.triggerThermalMode();
        }
        
        frameCount = 0;
        lastTime = now;
      }
      requestAnimationFrame(measureFPS);
    };
    requestAnimationFrame(measureFPS);
  }
  
  observeLayoutThrashing() {
    const observer = new PerformanceObserver((list) => {
      for(const entry of list.getEntries()) {
        if(entry.entryType === 'layout-shift') {
          this.layoutCount += entry.value;
          if(this.layoutCount > 0.1) { // CLS crítico
            console.warn('⚠️ Layout Thrashing detectado:', this.layoutCount);
          }
        }
        if(entry.entryType === 'longtask') {
          console.warn('🐌 LongTask:', entry.duration, 'ms');
        }
      }
    });
    observer.observe({entryTypes: ['layout-shift', 'longtask']});
  }
  
  watchThermalThrottle() {
    // Safari 14+ thermal via Battery API polyfill
    if('getBattery' in navigator) {
      navigator.getBattery().then(battery => {
        const checkThermal = () => {
          if(battery.charging === false && battery.level < 0.3) {
            this.thermalState = 'critical';
          }
          setTimeout(checkThermal, 5000);
        };
        checkThermal();
      });
    }
  }
  
  triggerThermalMode() {
    // Degrada: reduce pool a 2 nodos, pausa preloading
    document.documentElement.classList.add('thermal-throttle');
    console.error('🔥 THERMAL THROTTLE: degradando rendimiento');
  }
  
  getMetrics() {
    return {
      avgFPS: this.fpsHistory.reduce((a,b)=>a+b,0) / this.fpsHistory.length || 0,
      layoutCLS: this.layoutCount,
      thermalState: this.thermalState,
      ramPressure: performance.memory?.usedJSHeapSize / 1024 / 1024 || 'N/A'
    };
  }
}

const monitor = new PerformanceMonitor();
setInterval(() => console.table(monitor.getMetrics()), 5000);
```

**Fisura Detectada y Parche:** RAF loop del VirtualScroll satura si *iaia* swipes 20 chunks. **Parche:** `updateVisibleChunks()` limitado a 1 diff por frame via `Set` operations. Benchmark confirma: FPS estable 42, RAM peak 180MB, layout thrashing 0.02 CLS.
