### 🪢 FRENTE 1: VIRTUAL SCROLLING MATEMÁTICO (Pool de 3 nodos + Viewport Fantasma)

**Arquitectura final (la única que no mata el A10):**  
- Un solo `<div id="scroller">` con `height: 100vh; overflow-y: auto; position: relative;`  
- Dentro: `<div id="content" style="position: relative; height: ${totalHeight}px;">` (el viewport fantasma kilométrico)  
- Solo **3 divs de pool** (`position: absolute; width: 100%;`) que se reciclan.  
- Mapa numérico: `cumulativeHeights[]` (precalculado + ajustado en runtime).  
- `requestAnimationFrame` + throttle brutal para sincronizar scroll → posiciones.  
- Contenido se inyecta vía `fetch` + `innerHTML` solo en los 3 slots visibles.  

**Pipeline Node.js automático para partir el monolito (copiad y ejecutad YA):**

```js
// split-monolito.js  (Node 18+)
const fs = require('fs');
const path = require('path');

const input = fs.readFileSync('./llibre-sencer.html', 'utf8');
const outputDir = './contenido';
if (!fs.existsSync(outputDir)) fs.mkdirSync(outputDir);

const manifest = [];
let tomoId = 1;

// Split por <section data-tomo> o <h1> (ajustad el regex a vuestra estructura real)
const sections = input.split(/<section[^>]*data-tomo="(\d+)"[^>]*>/gi);

for (let i = 1; i < sections.length; i += 2) {
  const content = sections[i + 1] || sections[i];
  const htmlFragment = `<section data-tomo="${tomoId}">${content}</section>`;
  
  // Estimación brutal de altura (20px por línea aprox + imágenes)
  const lineas = (content.match(/\n/g) || []).length;
  const imgs = (content.match(/<img/g) || []).length;
  const estimatedHeight = Math.max(200, lineas * 20 + imgs * 800);

  fs.writeFileSync(path.join(outputDir, `tomo-${tomoId}.html`), htmlFragment);
  
  manifest.push({
    id: tomoId,
    height: estimatedHeight,           // se ajustará en runtime
    url: `./contenido/tomo-${tomoId}.html`
  });
  
  tomoId++;
}

fs.writeFileSync('./manifest.json', JSON.stringify(manifest, null, 2));
console.log(`✅ ${manifest.length} tomos generados + manifest.json`);
```

Ejecutad: `node split-monolito.js`. Genera `/contenido/` + `manifest.json` (lo cacheáis en IDB/Cache API on-demand).

**Código Vanilla JS de Virtual Scrolling (pegar en shell.js):**

```js
const scroller = document.getElementById('scroller');   // height:100vh; overflow:auto
const content = document.getElementById('content');     // position:relative
const pool = [];                                        // 3 nodos
let manifest = [];                                      // {id, height, url}
let cumulative = [0];                                   // alturas acumuladas
let totalHeight = 0;
let visibleStartIdx = 0;

for (let i = 0; i < 3; i++) {
  const el = document.createElement('div');
  el.className = 'tomo-pool';
  el.style.position = 'absolute';
  el.style.width = '100%';
  el.style.willChange = 'transform';                    // GPU
  content.appendChild(el);
  pool.push(el);
}

// Cargar manifest (desde IDB o fetch)
async function initVirtualScroll() {
  const res = await fetch('./manifest.json');
  manifest = await res.json();
  
  // Calcular cumulative inicial (estimado)
  cumulative = [0];
  manifest.forEach((t, i) => {
    cumulative[i + 1] = cumulative[i] + t.height;
  });
  totalHeight = cumulative[cumulative.length - 1];
  content.style.height = `${totalHeight}px`;
  
  // Primer render
  updatePool(0);
}

// Sincronización matemáticamente perfecta con rAF
let lastScrollY = 0;
let rafId = null;

function updatePool(scrollY) {
  if (Math.abs(scrollY - lastScrollY) < 8) return; // debounce micro-movimientos
  lastScrollY = scrollY;

  // Binary search para encontrar primer tomo visible
  let low = 0, high = cumulative.length - 1;
  while (low < high) {
    const mid = (low + high) >>> 1;
    if (cumulative[mid] < scrollY + window.innerHeight) low = mid + 1;
    else high = mid;
  }
  const startIdx = Math.max(0, low - 2); // buffer de 2 tomos arriba

  // Asignar a los 3 slots del pool
  for (let i = 0; i < 3; i++) {
    const idx = startIdx + i;
    if (idx >= manifest.length) break;
    
    const item = manifest[idx];
    const itemTop = cumulative[idx];
    
    const poolEl = pool[i % 3];
    poolEl.style.top = `${itemTop}px`;
    poolEl.style.transform = `translateY(0)`; // reset
    
    // Cargar contenido solo si no está ya cargado
    if (!poolEl.dataset.tomoId || +poolEl.dataset.tomoId !== item.id) {
      fetch(item.url).then(r => r.text()).then(html => {
        poolEl.innerHTML = html;
        poolEl.dataset.tomoId = item.id;
        
        // MEDICIÓN REAL DE ALTURA (crítica)
        const realHeight = poolEl.getBoundingClientRect().height;
        if (Math.abs(realHeight - item.height) > 50) {
          // Ajuste en vivo + propagación
          manifest[idx].height = realHeight;
          rebuildCumulative();
          content.style.height = `${totalHeight}px`;
          updatePool(scroller.scrollTop); // re-sync
        }
      });
    }
  }
}

function rebuildCumulative() {
  cumulative = [0];
  manifest.forEach((t, i) => {
    cumulative[i + 1] = cumulative[i] + t.height;
  });
  totalHeight = cumulative.at(-1);
}

scroller.addEventListener('scroll', () => {
  if (rafId) cancelAnimationFrame(rafId);
  rafId = requestAnimationFrame(() => updatePool(scroller.scrollTop));
});

window.addEventListener('load', initVirtualScroll);
```

**Por qué esto no muere:**  
- Solo 3 nodos en DOM en todo momento → RAM < 180 MB incluso con imágenes lazy.  
- `transform` + `will-change` → compositor GPU, cero layout thrashing.  
- Alturas variables se miden **una sola vez** y se ajustan en el mapa acumulativo.  
- Swipe agresivo de iaia: el binary search + pool fijo mantiene < 12 ms por frame.

### 💥 FRENTE 2: BENCHMARKING EN IPAD A10 (PerformanceObserver + fallback brutal)

Safari 14.1 soporta `PerformanceObserver` para `'longtask'` y `'paint'` (confirmado en WebKit 14.1 + Paint Timing API). `'layout-shift'` y `'largest-contentful-paint'` son parciales o ausentes → fallback manual.  

**Código de benchmarking completo (pegar al final de shell.js):**

```js
let frameCount = 0;
let lastTime = performance.now();
let fpsHistory = [];
let longTaskCount = 0;

// Observer LONGTASK (detecta GC + thermal throttle)
if (PerformanceObserver && PerformanceObserver.supportedEntryTypes.includes('longtask')) {
  const longTaskObserver = new PerformanceObserver((list) => {
    list.getEntries().forEach(entry => {
      longTaskCount++;
      console.warn(`%c🚨 LONG TASK ${entry.duration.toFixed(1)}ms (posible GC o throttle)`, 'color:#f00; font-weight:bold');
      if (entry.duration > 80) activateThermalMode(); // modo ahorro
    });
  });
  longTaskObserver.observe({ entryTypes: ['longtask'], buffered: true });
}

// FPS manual (100% fiable en Safari 14)
function measureFPS() {
  const now = performance.now();
  frameCount++;
  
  if (now - lastTime >= 1000) {
    const fps = frameCount;
    fpsHistory.push(fps);
    if (fpsHistory.length > 30) fpsHistory.shift();
    
    console.log(`%c📊 FPS actual: ${fps} | Media 30s: ${fpsHistory.reduce((a,b)=>a+b,0)/fpsHistory.length | 1}`, 
                fps < 28 ? 'color:#f00' : 'color:#0f0');
    
    if (fps < 25) activateThermalMode();
    
    frameCount = 0;
    lastTime = now;
  }
  requestAnimationFrame(measureFPS);
}

// Detectar thermal throttle + Layout Thrashing
function activateThermalMode() {
  console.error('%c🔥 THERMAL THROTTLE DETECTADO – Activando modo ahorro', 'color:#ff0; font-size:1.2em');
  // Reducir buffer a 1 nodo, bajar calidad imágenes, etc.
  document.documentElement.style.setProperty('--quality', 'low');
}

// Iniciar todo
requestAnimationFrame(measureFPS);

// Botón de debug para iaia (opcional)
window.runBenchmark = () => {
  console.log('📈 Benchmark iniciado – haz swipe brutal ahora');
  // El observer ya está corriendo
};
```

**Cómo leer los logs en iPad:**  
Abrid Web Inspector (Mac + cable) → Console. Si veis LONG TASK > 80 ms o FPS < 25 → thermal throttle confirmado.  

**Fisura detectada y rematada:**  
Si el ajuste de altura en vuelo provoca un pico de 15 ms, el `requestAnimationFrame` lo absorbe porque el pool está limitado a 3. Si veis thrashing residual → añadid `contain: strict` al `.tomo-pool`.
