### 2. PREFIJO ACUMULADO (MAPA DE ALTURAS)

```js
const offsets = []; // posición Y de cada item

function buildOffsets() {
  let acc = 0;
  for (let i = 0; i < items.length; i++) {
    offsets[i] = acc;
    acc += items[i].height || 800; // fallback estimado
  }
  totalHeight = acc;
}
```

## 🔍 BÚSQUEDA BINARIA (CLAVE)

```js
function findIndex(y) {
  let low = 0, high = offsets.length - 1;

  while (low <= high) {
    const mid = (low + high) >> 1;

    if (offsets[mid] <= y && offsets[mid + 1] > y) {
      return mid;
    }

    if (offsets[mid] < y) low = mid + 1;
    else high = mid - 1;
  }

  return 0;
}
```

## 🔄 UPDATE LOOP

```js
const nodes = [
  document.getElementById('node0'),
  document.getElementById('node1'),
  document.getElementById('node2')
];

function update() {
  const baseIndex = findIndex(scrollY);

  for (let i = 0; i < 3; i++) {
    const itemIndex = baseIndex + i;
    const node = nodes[i];
    const item = items[itemIndex];

    if (!item) continue;

    node.style.transform = `translateY(${offsets[itemIndex]}px)`;

    if (node.dataset.id !== item.id) {
      renderNode(node, item);
    }
  }

  ticking = false;
}
```

## 🧠 1. MEDICIÓN FPS (CRÍTICA)

```js
let lastFrame = performance.now();
let fps = 60;

function monitorFPS() {
  requestAnimationFrame(now => {
    const delta = now - lastFrame;
    fps = 1000 / delta;

    lastFrame = now;

    monitorFPS();
  });
}

monitorFPS();
```
