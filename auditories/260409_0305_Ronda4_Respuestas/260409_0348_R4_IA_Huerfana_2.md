Si inyectamos un mapa fijo, el scroll inercial colapsará en abismos blancos.

Aquí tenéis la reconstrucción quirúrgica: Topología heurística en el servidor, anclaje asíncrono matemático en el cliente y telemetría proxy para el infarto térmico.

### 🪢 FRENTE 1: VIRTUAL SCROLLING MATEMÁTICO (El Anillo de 3 Nodos)

#### 1. El Oráculo Ciego (Pipeline Node.js)
El servidor destruye el monolito y genera una estimación asintótica. Esto proporciona el esqueleto espacial para que el cliente pueda pintar la barra de scroll nativa antes de descargar un solo byte de texto.

```javascript
// pipeline-guillotina.js (Ejecución en fase Build)
const fs = require('fs');

function forjarGenotipo(htmlMonolitico) {
  const tomos = htmlMonolitico.split(/(?=<article class="tomo-sdp">)/i);
  let offsetY = 0;
  const mapaVirtual = [];

  tomos.forEach((html, i) => {
    // Heurística de Trellat rural (Text-size: GEM MODERN)
    const numChars = html.replace(/<[^>]+>/g, '').length;
    const numImgs = (html.match(/<img/g) || []).length;
    
    // Estimamos ~0.4px de altura por carácter + 300px por imagen
    const altoEstimado = Math.floor(numChars * 0.4) + (numImgs * 300) + 150;

    mapaVirtual.push({ id: i, top: offsetY, h: altoEstimado, medido: false });
    offsetY += altoEstimado;
    
    // El Genotipo se astilla en JSONs listos para IndexedDB
    fs.writeFileSync(`./dist/chunks/tomo_${i}.json`, JSON.stringify({ html }));
  });

  fs.writeFileSync('./dist/mapa-genotipo.json', JSON.stringify({ total: offsetY, mapa: mapaVirtual }));
}
```

#### 2. La Aritmética del Vacío (Motor Vanilla JS en Cliente)
Safari 14 nos ahogará si usamos `appendChild` o `remove`. El DOM queda reducido a **3 únicos nodos inmortales** con `position: absolute`. Para reasignarlos sin colisiones en el espacio/tiempo mientras la usuaria hace scroll, usamos el operador módulo (`idx % 3`).

```javascript
// core-motor-rural.js (Hardware: iPad A10)
class MotorRuralA10 {
  constructor(datosGenotipo) {
    this.mapa = datosGenotipo.mapa;
    this.totalH = datosGenotipo.total;
    
    // El fantasma que fuerza la barra de scroll nativa de iOS
    this.abismo = document.getElementById('abismo-scroll'); 
    this.abismo.style.height = `${this.totalH}px`;

    // El Anillo de VRAM
    this.pool = [this.forjarNodo(), this.forjarNodo(), this.forjarNodo()];
    this.indiceActual = -1;
    this.ticking = false;

    // EL CORTAFUEGOS DEL LAYOUT SHIFT
    this.observador = new ResizeObserver((entradas) => this.corregirGeometriaAsincrona(entradas));
    this.pool.forEach(nodo => this.observador.observe(nodo));

    window.addEventListener('scroll', () => this.onScroll(), { passive: true });
  }

  forjarNodo() {
    const el = document.createElement('div');
    // Aislamiento termodinámico estricto: El navegador jamás propaga cambios de aquí al Body.
    el.style.cssText = 'position: absolute; top: 0; left: 0; width: 100%; contain: strict; will-change: transform; visibility: hidden;';
    document.getElementById('visor-visor').appendChild(el);
    return el;
  }

  // Complejidad O(log N): Buscar en un libro de 2.000 fragmentos cuesta 11 iteraciones (<0.1ms)
  buscarIndiceBinario(y) {
    let L = 0, R = this.mapa.length - 1;
    while (L <= R) {
      let M = Math.floor((L + R) / 2);
      let t = this.mapa[M];
      if (y >= t.top && y < t.top + t.h) return M;
      if (y < t.top) R = M - 1;
      else L = M + 1;
    }
    return 0;
  }

  onScroll() {
    if (!this.ticking) {
      requestAnimationFrame(() => {
        this.renderizar(window.scrollY);
        this.ticking = false;
      });
      this.ticking = true;
    }
  }

  async renderizar(scrollY) {
    const idx = this.buscarIndiceBinario(scrollY);
    if (idx === this.indiceActual) return;
    this.indiceActual = idx;

    const ventanaActiva = [idx - 1, idx, idx + 1].filter(i => i >= 0 && i < this.mapa.length);

    // Purga de colisiones visuales residuales
    this.pool.forEach(n => {
      const idNodo = parseInt(n.dataset.idx);
      if (!ventanaActiva.includes(idNodo)) n.style.visibility = 'hidden';
    });

    for (let i of ventanaActiva) {
      const nodo = this.pool[i % 3]; // Rotación Cíclica Inquebrantable
      const meta = this.mapa[i];

      if (nodo.dataset.idx !== String(i)) {
        nodo.dataset.idx = i;
        
        // 1. Traslación directa a la GPU de Apple (Bypass del Hilo de Renderizado)
        nodo.style.transform = `translate3d(0, ${meta.top}px, 0)`;
        nodo.style.visibility = 'visible';
        
        // 2. Extracción O(1) vía IndexedDB + Caché en Memoria Local
        const { html } = await idbMestra.getChunk(`tomo_${i}`);
        nodo.innerHTML = html;
        // Al inyectar el HTML, el ResizeObserver despertará si Node.js mintió sobre la altura.
      }
    }
  }

  corregirGeometriaAsincrona(entradas) {
    let ajusteScrollInverso = 0;
    
    for (let entrada of entradas) {
      const nodo = entrada.target;
      const idx = parseInt(nodo.dataset.idx);
      if (isNaN(idx)) continue;

      const meta = this.mapa[idx];
      // Safari 14 soporta contentRect. Evitamos offsetHeight para no invocar reflow forzado.
      const alturaReal = entrada.contentRect.height;
      const delta = alturaReal - meta.h;

      if (Math.abs(delta) > 5) {
        meta.h = alturaReal;
        meta.medido = true;

        // Desplazar el mapa inferior O(N) -> <1ms en Vanilla JS para arrays numéricos puros
        for (let j = idx + 1; j < this.mapa.length; j++) {
          this.mapa[j].top += delta;
        }
        this.totalH += delta;
        this.abismo.style.height = `${this.totalH}px`;

        // SCROLL ANCHORING ABSOLUTO: Si el fragmento mutado está por ENCIMA del
        // campo visual de la iaia, el texto actual saltará hacia abajo y ella perderá su lectura.
        // Lo neutralizamos inyectando la misma cantidad de píxeles al scroll nativo al instante.
        if (meta.top < window.scrollY) {
          ajusteScrollInverso += delta;
        }
      }
    }

    if (ajusteScrollInverso !== 0) {
      window.scrollBy({ top: ajusteScrollInverso, behavior: 'instant' });
    }
  }
}
```

### 💥 FRENTE 2: LA MATRIZ DE SUPERVIVENCIA (Telemetría de Bajo Nivel)

Safari 14 es una caja negra. No expone APIs modernas de rendimiento (`performance.memory`, `layout-shift`). Si queremos cazar el OOM (Out Of Memory) antes de que *Jetsam* fusile la PWA y nos deje una pantalla blanca, tenemos que instrumentar la latencia residual del Event Loop y someter a *Monkey Patching* el DOM.

Esta es la Sonda del A10.

```javascript
// watchdog-termico.js
const SondaA10 = (function() {
  let ultimoLatidoRAF = performance.now();
  let asfixiaCPU = 0;
  let modoSupervivencia = false;

  function iniciar() {
    interceptarLayoutThrashing();
    requestAnimationFrame(monitorFrameRate);
    monitorGarbageCollector();
  }

  // 1. EL CAZADOR DE REFLOWS SÍNCRONOS
  function interceptarLayoutThrashing() {
    let lecturasEnEsteFrame = 0;
    const originBCR = Element.prototype.getBoundingClientRect;
    
    Element.prototype.getBoundingClientRect = function() {
      lecturasEnEsteFrame++;
      // Si el código lee geometría 3 veces sin ceder al Paint, la CPU está atrapada en Thrashing.
      if (lecturasEnEsteFrame > 3) {
        console.error("🩸 [RED TEAM ALERTA] Layout Thrashing Masivo detectado en Frame Actual.");
      }
      return originBCR.apply(this, arguments);
    };

    requestAnimationFrame(function reset() {
      lecturasEnEsteFrame = 0;
      requestAnimationFrame(reset);
    });
  }

  // 2. EL RADAR TÉRMICO (Detección de Caída de FPS)
  function monitorFrameRate(now) {
    const delta = now - ultimoLatidoRAF;
    ultimoLatidoRAF = now;

    // Un frame debe tardar 16.6ms. Si sobrepasa los 33ms de forma constante,
    // el procesador se está sobrecalentando (Thermal Throttling de Apple).
    if (delta > 33) {
      asfixiaCPU++;
      // 15 frames ahogados = Activación del Búnker
      if (asfixiaCPU > 15 && !modoSupervivencia) activarThermalThrottle("GPU Throttled (Frame Drop Constante)");
    } else {
      // Enfriamiento matemático
      asfixiaCPU = Math.max(0, asfixiaCPU - 0.2);
    }
    requestAnimationFrame(monitorFrameRate);
  }

  // 3. EL SISMÓGRAFO DE MEMORIA (Detección de Bloqueos del Main Thread)
  function monitorGarbageCollector() {
    let ultimoTick = performance.now();
    
    // Una macro-task espaciada cada 100ms. Si el Garbage Collector de JS entra
    // en pánico para borrar miles de strings de la RAM, este Timer se desfasará brutalmente.
    setInterval(() => {
      const now = performance.now();
      const drift = now - ultimoTick;
      ultimoTick = now;
      
      if (drift > 220 && !modoSupervivencia) {
        activarThermalThrottle(`Main Thread Bloqueado / GC Spike Severo: ${drift.toFixed(0)}ms`);
      }
    }, 100);
  }

  function activarThermalThrottle(motivo) {
    modoSupervivencia = true;
    console.warn(`🔥 [A10 ALERTA TÉRMICA] Degradando VRAM y Render. Motivo: ${motivo}`);
    
    // 1. Simplificación extrema del Árbol de Renderizado nativo
    document.documentElement.classList.add('sdp-modo-supervivencia');
    
    // 2. Ejecutar guillotina asíncrona: Matar regex de highlight.js, frenar precargas.
    window.dispatchEvent(new Event('sdp-termico-abortar'));
  }

  return { iniciar };
})();

SondaA10.iniciar();
```
