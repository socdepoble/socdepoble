### Estrategia resumida
**Objetivo**: mantener el DOM en **3 nodos reciclables** que se reposicionan con `transform: translateY()` sobre un contenedor con altura virtual, evitar parseo masivo en main thread, controlar fetchs concurrentes con semáforos y AbortController, almacenar fragments grandes en IndexedDB por chunks y medir en cliente FPS, Long Tasks y memoria para activar `thermal_throttle_mode`. Las técnicas clave son virtual scroll absoluto, streaming parse, fetch queue con prioridad y chunked IDB writes.

### Virtual Scrolling Absoluto — implementación Vanilla JS (3 nodos reciclables)

**Idea**: un contenedor `#viewport` con `overflow:auto` y un `#spacer` que representa la altura total. Solo tres nodos DOM visibles se mantienen: `nodePrev`, `nodeCurr`, `nodeNext`. Al hacer scroll se calcula qué fragmento corresponde a la posición y se recicla el nodo más lejano actualizando su `transform: translateY()` y su contenido. Las alturas de fragmentos son variables y se mantienen en un mapa `heights[]` con estimados y correcciones tras medir.

**Principios**  
- **No usar top/absolute reflow** para cada elemento. Usar `transform` para mover nodos.  
- **Mantener spacers** para preservar offsets y evitar layout thrash.  
- **Calcular índice por acumulado de alturas** usando un árbol prefix‑sum para búsqueda O(log n).  
- **Reutilizar nodos** y actualizar contenido off‑DOM antes de swap.

**Código esencial (cliente)**

```html
<div id="viewport" style="height:100vh;overflow-y:auto;position:relative;">
  <div id="spacer" style="height:1000000px;position:relative;"></div>
</div>
```

```js
// Config
const POOL_SIZE = 3;
const viewport = document.getElementById('viewport');
const spacer = document.getElementById('spacer');

// Estado
const totalFragments = manifest.length;
const heights = new Float64Array(totalFragments); // estimated heights in px
const prefix = new Float64Array(totalFragments + 1); // prefix sums
let pool = []; // {el, fragmentId, top}

// Build prefix sums helper
function rebuildPrefix() {
  prefix[0] = 0;
  for (let i = 0; i < totalFragments; i++) prefix[i+1] = prefix[i] + (heights[i] || 800);
  spacer.style.height = prefix[totalFragments] + 'px';
}

// Binary search: find fragment index for scrollTop
function findIndex(scrollTop) {
  let lo = 0, hi = totalFragments;
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (prefix[mid+1] <= scrollTop) lo = mid + 1; else hi = mid;
  }
  return Math.max(0, lo);
}

// Create pool nodes
for (let i = 0; i < POOL_SIZE; i++) {
  const el = document.createElement('div');
  el.className = 'vs-node';
  el.style.position = 'absolute';
  el.style.left = '0';
  el.style.width = '100%';
  el.style.willChange = 'transform';
  pool.push({el, fragmentId: -1, top: 0});
  spacer.appendChild(el);
}

// Fetch queue instance provided elsewhere
const fetchQueue = new FetchQueue(2);

// Recycle logic
let lastRenderIndex = -1;
function onScrollRAF() {
  const scrollTop = viewport.scrollTop;
  const centerIndex = findIndex(scrollTop + viewport.clientHeight / 2);
  if (centerIndex === lastRenderIndex) return;
  lastRenderIndex = centerIndex;

  // Determine which fragment ids to show: prev, curr, next
  const ids = [
    Math.max(0, centerIndex - 1),
    centerIndex,
    Math.min(totalFragments - 1, centerIndex + 1)
  ];

  // Map pool slots to ids, reuse existing if present
  const used = new Set(ids);
  for (let i = 0; i < POOL_SIZE; i++) {
    const slot = pool[i];
    const targetId = ids[i];
    if (slot.fragmentId === targetId) {
      // already correct
      const top = prefix[targetId];
      slot.top = top;
      slot.el.style.transform = `translateY(${top}px)`;
      continue;
    }
    // Recycle slot: fetch fragment and inject off-DOM
    slot.fragmentId = targetId;
    const top = prefix[targetId];
    slot.top = top;
    slot.el.style.transform = `translateY(${top}px)`;
    // Load content via fetchQueue
    fetchQueue.enqueue(signal => fetchFragment(targetId, {signal}))
      .then(html => {
        // inject safely off-DOM
        const df = document.createRange().createContextualFragment(html);
        // measure height off-DOM
        const temp = document.createElement('div');
        temp.style.position = 'absolute';
        temp.style.visibility = 'hidden';
        temp.appendChild(df.cloneNode(true));
        document.body.appendChild(temp);
        const measured = temp.offsetHeight || (heights[targetId] || 800);
        document.body.removeChild(temp);
        heights[targetId] = measured;
        rebuildPrefix();
        // final inject
        slot.el.innerHTML = html;
        slot.el.style.transform = `translateY(${prefix[targetId]}px)`;
        scheduleHighlightForNode(slot.el);
      })
      .catch(()=>{ /* handle abort */ });
  }
}

// Centralized rAF loop
let rafId = null;
function rafLoop(ts) {
  onScrollRAF();
  rafId = requestAnimationFrame(rafLoop);
}
rafId = requestAnimationFrame(rafLoop);

// Attach scroll event to throttle updates
viewport.addEventListener('scroll', () => {
  // throttle to rAF only
});
```

**Notas**  
- El `findIndex` usa prefix sums para localizar fragmento por offset en O(log n). Mantener `heights` actualizados reduce saltos.

### Pipeline Node.js para partir el monolito y estimar alturas

**Objetivo**: dividir `llibre-sencer.html` en `fragments/0001.html` y generar `manifest.json` con `estimatedHeight` usando Puppeteer para render real y medir altura. Esto evita estimados pobres y alimenta el spacer con valores reales.

**Script Node (esqueleto)**

```js
// split-and-measure.js
const fs = require('fs');
const path = require('path');
const puppeteer = require('puppeteer');

async function splitAndMeasure(inputHtml, outDir) {
  const html = fs.readFileSync(inputHtml,'utf8');
  // split markers: <!-- FRAGMENT id:title -->
  const parts = html.split(/<!--\s*FRAGMENT\s+([^\s:]+):([^\-]+)\s*-->/g);
  // parts[0] is preamble, then groups of id,title,content...
  const fragments = [];
  for (let i = 1; i < parts.length; i += 3) {
    const id = parts[i].trim();
    const title = parts[i+1].trim();
    const content = parts[i+2];
    fragments.push({id, title, content});
  }
  if (!fs.existsSync(outDir)) fs.mkdirSync(outDir);
  const browser = await puppeteer.launch({args:['--no-sandbox']});
  const page = await browser.newPage();
  const manifest = [];
  for (const f of fragments) {
    const file = path.join(outDir, `${f.id}.html`);
    fs.writeFileSync(file, f.content, 'utf8');
    await page.setContent(f.content, {waitUntil:'networkidle2'});
    const height = await page.evaluate(()=> document.body.scrollHeight);
    manifest.push({id:f.id, title:f.title, url:`/fragments/${f.id}.html`, estimatedHeight: height});
  }
  await browser.close();
  fs.writeFileSync(path.join(outDir,'manifest.json'), JSON.stringify(manifest, null, 2));
}

splitAndMeasure('llibre-sencer.html','fragments').catch(console.error);
```

**Resultado**: `manifest.json` con `estimatedHeight` para cada fragmento que alimenta `heights[]` iniciales. Esto reduce reflows al rehidratar.

### FetchQueue, cancelación y rehidratación segura

**FetchQueue** controla concurrencia, prioridad y AbortController. Cancela fetches en cola si el usuario swypea demasiado rápido. Implementar hysteresis en IntersectionObserver y gating por scroll velocity.

```js
class FetchQueue {
  constructor(concurrency=2){
    this.concurrency = concurrency;
    this.running = 0;
    this.queue = [];
  }
  enqueue(task, priority=5){
    return new Promise((res, rej) => {
      this.queue.push({task, priority, res, rej});
      this.queue.sort((a,b)=> b.priority - a.priority);
      this.dequeue();
    });
  }
  dequeue(){
    if (this.running >= this.concurrency) return;
    const item = this.queue.shift();
    if (!item) return;
    this.running++;
    const ctrl = new AbortController();
    item.task(ctrl.signal).then(r => {
      this.running--;
      item.res(r);
      this.dequeue();
    }).catch(e => {
      this.running--;
      item.rej(e);
      this.dequeue();
    });
  }
}
```

**Rehidratación atómica**  
- Parsear off‑DOM con `createContextualFragment`.  
- Medir altura en nodo temporal oculto.  
- Reemplazar spacer con `requestAnimationFrame` en una sola operación. Esto evita layout thrash.

### Benchmarking en iPad A10 — instrumentación cliente

**Objetivo**: detectar jank, long tasks y señales de memoria para activar `thermal_throttle_mode`. Safari 14 limita algunas APIs pero PerformanceObserver y Long Task API pueden estar disponibles. Usar `performance.now`, `requestAnimationFrame` y `PerformanceObserver` para medir FPS y long tasks. Usar `performance.memory` si existe y `navigator.deviceMemory` como hint.

**Instrumentos clave**

1. **FPS monitor** con rAF delta
```js
class FPSMeter {
  constructor(sampleWindow = 1000){
    this.last = performance.now();
    this.frames = 0;
    this.start = this.last;
    this.fps = 60;
    this.sampleWindow = sampleWindow;
    this.loop = this.loop.bind(this);
    requestAnimationFrame(this.loop);
  }
  loop(ts){
    this.frames++;
    const elapsed = ts - this.start;
    if (elapsed >= this.sampleWindow) {
      this.fps = (this.frames / elapsed) * 1000;
      this.frames = 0;
      this.start = ts;
      // report fps
      window.dispatchEvent(new CustomEvent('perf:fps', {detail:{fps:this.fps}}));
    }
    requestAnimationFrame(this.loop);
  }
}
```

2. **Long Task observer**
```js
let longTaskCount = 0;
if (PerformanceObserver && PerformanceObserver.supportedEntryTypes && PerformanceObserver.supportedEntryTypes.includes('longtask')) {
  const obs = new PerformanceObserver(list => {
    for (const entry of list.getEntries()) {
      longTaskCount++;
      window.dispatchEvent(new CustomEvent('perf:longtask', {detail:{duration:entry.duration}}));
    }
  });
  obs.observe({entryTypes:['longtask']});
}
```

3. **Heap and memory heuristics**  
- Use `performance.memory.usedJSHeapSize` if available. Fallback to heuristic allocations and measuring GC pauses is risky on production. Use `navigator.deviceMemory` to set conservative budgets.

4. **Thermal throttle trigger logic**
```js
const thresholds = {
  fpsLow: 30,
  longTaskRate: 3, // long tasks per minute
  heapPeak: 200 * 1024 * 1024 // 200MB
};
let longTaskWindow = [];
window.addEventListener('perf:longtask', e => {
  longTaskWindow.push(Date.now());
  longTaskWindow = longTaskWindow.filter(t => t > Date.now() - 60000);
  if (longTaskWindow.length > thresholds.longTaskRate) activateThermalThrottle();
});
window.addEventListener('perf:fps', e => {
  if (e.detail.fps < thresholds.fpsLow) activateThermalThrottle();
});
function checkHeap(){
  if (performance.memory && performance.memory.usedJSHeapSize > thresholds.heapPeak) activateThermalThrottle();
}
function activateThermalThrottle(){
  // set global flags to reduce concurrency, disable prefetch, reduce pool to 1, defer highlights
  window.__thermalThrottle = true;
  fetchQueue.concurrency = 1;
  // prune to current fragment only
  pruneToCurrent();
}
```

**Medición de GC y long tasks en Safari**  
- Safari no expone GC events. Medir long tasks via PerformanceObserver es la mejor señal. Complementar con FPS drops y `usedJSHeapSize` cuando esté disponible.

### Checklist operativo y thresholds para iPad A10

- **DOM nodes**: máximo 3 visibles.  
- **Fetch concurrency**: 1–2 en low memory. Default 2.  
- **Prefetch budget**: 2 tokens regenerando 1 token cada 3s.  
- **Heap peak objetivo**: < 200MB. Si `performance.memory` no disponible, usar conservative deviceMemory <=2 → assume low memory.  
- **rAF budget**: mantener trabajo por frame < 8ms para dejar margen al compositor. Si frame work > 8ms, degradar.  
- **Highlighting**: WebWorker pipeline; fallback idle chunking.  
- **Storage**: hybrid Cache for shell and small fragments; IndexedDB chunked writes for large fragments and images. No `caches.addAll` masivo en install. Implementar LRU in IDB.
