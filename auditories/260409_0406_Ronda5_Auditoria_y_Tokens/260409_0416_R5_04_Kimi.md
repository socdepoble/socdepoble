### VECTOR 1: JETSAM DE INDEXEDDB EN SAFARI 14 (El Asesino Silencioso)

**La fisura:** Safari en iOS 14 implementa Jetsam (OOM killer agresivo). IndexedDB no tiene garantía de persistencia bajo presión de memoria. Los datos pueden evaporarse sin notificación cuando el sistema necesita RAM para otras apps.

**Ataque simulado:**
```javascript
// Simulación de presión de memoria que activa Jetsam
const memoryBomb = () => {
  const arrays = [];
  while (true) {
    arrays.push(new Uint8Array(1024 * 1024)); // 1MB chunks
    if (performance.memory?.usedJSHeapSize > 1500000000) break;
  }
  return arrays;
};
// Resultado: Safari mata el proceso sin evento 'beforeunload' confiable
```

**Solución - Sistema de Checkpoint Híbrido con Crypto-Hash:**
```javascript
/**
 * GENOTIPO_CHECKPOINT: Sistema de respaldo immutable contra Jetsam
 * Cada chunk se almacena con hash SHA-256 para detectar corrupción/evaporación
 */

class JetsamResistantStore {
  constructor() {
    this.idb = null;
    this.checkpointStore = 'checkpoints_v1';
    this.hashCache = new Map(); // LRU de hashes verificados
    this.lastCheckpoint = 0;
    this.checkpointInterval = 30000; // 30s
    this.init();
  }

  async init() {
    this.idb = await openDB('genotype-immune', 1, {
      upgrade(db) {
        // Store principal
        db.createObjectStore('chunks', { keyPath: 'id' });
        // Store de checkpoints (solo metadatos, pequeño)
        db.createObjectStore('checkpoints', { keyPath: 'timestamp' });
        // Store de hashes (para verificación rápida)
        db.createObjectStore('hashes', { keyPath: 'chunkId' });
      }
    });
    
    // Verificación de integridad al arrancar
    await this.verifyIntegrity();
    
    // Checkpoint periódico
    setInterval(() => this.createCheckpoint(), this.checkpointInterval);
    
    // Escuchar visibilidad (app backgrounded = riesgo de Jetsam)
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) this.createCheckpoint();
    });
  }

  async computeHash(data) {
    const json = JSON.stringify(data);
    const encoder = new TextEncoder();
    const buffer = encoder.encode(json);
    
    // SubtleCrypto disponible en Safari 14
    const hashBuffer = await crypto.subtle.digest('SHA-256', buffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  }

  async storeChunk(id, data) {
    const hash = await this.computeHash(data);
    
    const transaction = this.idb.transaction(['chunks', 'hashes'], 'readwrite');
    
    // Guardar chunk
    await transaction.objectStore('chunks').put({
      id,
      data,
      storedAt: Date.now(),
      attempts: 0
    });
    
    // Guardar hash separado (más pequeño, más seguro)
    await transaction.objectStore('hashes').put({
      chunkId: id,
      hash,
      timestamp: Date.now()
    });
    
    this.hashCache.set(id, hash);
    await transaction.done;
  }

  async verifyIntegrity() {
    const hashStore = this.idb.transaction('hashes').objectStore('hashes');
    const allHashes = await hashStore.getAll();
    
    const corrupted = [];
    
    for (const { chunkId, hash: expectedHash } of allHashes) {
      const chunk = await this.idb.get('chunks', chunkId);
      
      if (!chunk) {
        // EVAPORACIÓN DETECTADA (Jetsam)
        corrupted.push({ chunkId, reason: 'EVAPORATED' });
        continue;
      }
      
      const actualHash = await this.computeHash(chunk.data);
      if (actualHash !== expectedHash) {
        // CORRUPCIÓN DETECTADA
        corrupted.push({ chunkId, reason: 'CORRUPTED' });
      }
    }
    
    if (corrupted.length > 0) {
      await this.recoverFromCorruption(corrupted);
    }
    
    return corrupted.length === 0;
  }

  async recoverFromCorruption(corruptedList) {
    // Intentar recuperar desde Cache API
    const cache = await caches.open('genotype-v4');
    
    for (const { chunkId, reason } of corruptedList) {
      console.warn(`[JETSAM] Recuperando chunk ${chunkId} (${reason})`);
      
      // Eliminar entrada corrupta
      await this.idb.delete('chunks', chunkId);
      await this.idb.delete('hashes', chunkId);
      
      // Re-fetch desde red/cache
      const response = await cache.match(`/chunks/chunk-${chunkId}.json`);
      if (response) {
        const data = await response.json();
        await this.storeChunk(chunkId, data);
      }
    }
  }

  async createCheckpoint() {
    const checkpoint = {
      timestamp: Date.now(),
      chunkCount: (await this.idb.getAllKeys('chunks')).length,
      hashCount: (await this.idb.getAllKeys('hashes')).length,
      scrollPosition: window.scrollY,
      userSession: this.getSessionId()
    };
    
    await this.idb.put('checkpoints', checkpoint);
    this.lastCheckpoint = checkpoint.timestamp;
  }

  getSessionId() {
    // Identificador de sesión persistente en sessionStorage
    let id = sessionStorage.getItem('genotype-session');
    if (!id) {
      id = crypto.randomUUID();
      sessionStorage.setItem('genotype-session', id);
    }
    return id;
  }
}
```

### VECTOR 2: CACHE POISONING Y CORRUPCIÓN DE PWA

**La fisura:** La Cache API puede ser envenenada por interceptores, actualizaciones fallidas, o storage quota exceeded. Un Service Worker mal actualizado puede dejar la app en estado zombie.

**Ataque:** Service Worker actualiza pero el nuevo cache no se activa. Usuario ve versión antigua corrupta indefinidamente.

**Solución - Sistema de Activación Atómica con Rollback:**
```javascript
// service-worker.js - GENOTIPO SW v4.1

const CACHE_VERSION = 'genotype-v4.1';
const CACHE_PREVIOUS = 'genotype-v4.0';
const MANIFEST_URL = '/genotype-manifest.json';

// Estrategia: Install con verificación de integridad
self.addEventListener('install', (event) => {
  event.waitUntil(
    (async () => {
      // 1. Descargar nuevo manifest
      const manifestResponse = await fetch(MANIFEST_URL);
      const manifest = await manifestResponse.json();
      
      // 2. Verificar checksums (si están en manifest)
      const cache = await caches.open(CACHE_VERSION);
      const criticalAssets = [
        '/index.html',
        '/genotype-scroller.js',
        '/genotype-thermal.js',
        ...manifest.chunks.map(c => c.path)
      ];
      
      // 3. Descarga atómica: todo o nada
      const fetchPromises = criticalAssets.map(async (url) => {
        const response = await fetch(url, { cache: 'no-cache' });
        
        // Verificar que no sea 404 o error
        if (!response.ok) {
          throw new Error(`CRITICAL: Failed to fetch ${url}`);
        }
        
        // Verificar Content-Type para JS
        if (url.endsWith('.js')) {
          const contentType = response.headers.get('content-type');
          if (!contentType || !contentType.includes('javascript')) {
            throw new Error(`CORRUPTION: ${url} no es JS válido`);
          }
        }
        
        return cache.put(url, response);
      });
      
      // Si cualquier fetch falla, abortar instalación
      await Promise.all(fetchPromises);
      
      // 4. Marcar como "listo para activación"
      await cache.put('/.genotype-ready', new Response('READY'));
      
      self.skipWaiting();
    })()
  );
});

// Activación atómica con fallback
self.addEventListener('activate', (event) => {
  event.waitUntil(
    (async () => {
      const cache = await caches.open(CACHE_VERSION);
      const ready = await cache.match('/.genotype-ready');
      
      if (!ready) {
        // Instalación incompleta, abortar
        console.error('[SW] Instalación corrupta detectada');
        await caches.delete(CACHE_VERSION);
        return;
      }
      
      // Limpiar caches antiguos solo si nuevo está sano
      const cacheNames = await caches.keys();
      await Promise.all(
        cacheNames
          .filter(name => name !== CACHE_VERSION)
          .map(name => caches.delete(name))
      );
      
      await self.clients.claim();
    })()
  );
});

// Fetch con stale-while-revalidate y detección de corrupción
self.addEventListener('fetch', (event) => {
  const { request } = event;
  
  // Solo interceptar requests de la app
  if (!request.url.includes(self.location.origin)) return;
  
  event.respondWith(
    (async () => {
      const cache = await caches.open(CACHE_VERSION);
      const cached = await cache.match(request);
      
      // Estrategia: Cache First + Background Revalidate
      if (cached) {
        // Revalidar en background (no bloquear)
        event.waitUntil(
          fetch(request)
            .then(response => {
              if (response.ok) {
                cache.put(request, response.clone());
              }
            })
            .catch(() => {}) // Ignorar errores de red en background
        );
        
        return cached;
      }
      
      // No en cache: fetch de red
      try {
        const networkResponse = await fetch(request);
        if (networkResponse.ok) {
          cache.put(request, networkResponse.clone());
        }
        return networkResponse;
      } catch (error) {
        // Offline y no en cache: devolver fallback
        if (request.destination === 'document') {
          return cache.match('/offline-fallback.html');
        }
        return new Response('Offline', { status: 503 });
      }
    })()
  );
});

// Escuchar mensajes de limpieza de emergencia
self.addEventListener('message', (event) => {
  if (event.data === 'EMERGENCY_PURGE') {
    caches.keys().then(names => 
      Promise.all(names.map(name => caches.delete(name)))
    );
  }
});
```

### VECTOR 3: ESTRANGULAMIENTO EN SINCRONIZACIÓN P2P OFFLINE

**La fisura:** Cuando dos dispositivos rurales intentan sincronizar datos offline (Bluetooth/WiFi Direct), la falta de coordinación causa conflictos irreconciliables o duplicación masiva.

**Solución - CRDTs (Conflict-free Replicated Data Types) ligero:**
```javascript
/**
 * GENOTIPO_SYNC: CRDT simplificado para sincronización rural P2P
 * Basado en LWW-Element-Set (Last-Write-Wins) con vector clocks
 */

class GenotypeCRDT {
  constructor(nodeId) {
    this.nodeId = nodeId;
    this.vectorClock = new Map(); // { nodeId: timestamp }
    this.data = new Map(); // { key: { value, timestamp, nodeId } }
    this.tombstones = new Set(); // Eliminaciones
  }

  // Incrementar reloj vectorial local
  tick() {
    const current = this.vectorClock.get(this.nodeId) || 0;
    this.vectorClock.set(this.nodeId, current + 1);
    return this.vectorClock.get(this.nodeId);
  }

  // Añadir/Actualizar
  set(key, value) {
    const timestamp = Date.now();
    this.tick();
    
    this.data.set(key, {
      value,
      timestamp,
      nodeId: this.nodeId,
      vectorClock: new Map(this.vectorClock)
    });
    
    // Eliminar de tombstones si existía
    this.tombstones.delete(key);
  }

  // Eliminar (soft delete con tombstone)
  delete(key) {
    this.tick();
    this.tombstones.add(key);
    this.data.delete(key);
  }

  // Merge con otro nodo (sincronización P2P)
  merge(otherCRDT) {
    const conflicts = [];
    
    // Fusionar datos
    for (const [key, theirEntry] of otherCRDT.data) {
      const ourEntry = this.data.get(key);
      
      if (!ourEntry) {
        // No tenemos esta clave: adoptar
        this.data.set(key, theirEntry);
      } else {
        // Conflicto: resolver con LWW + desempate por nodeId
        const resolution = this.resolveConflict(ourEntry, theirEntry);
        
        if (resolution === 'theirs') {
          conflicts.push({ key, ours: ourEntry, theirs: theirEntry });
          this.data.set(key, theirEntry);
        }
      }
    }
    
    // Fusionar tombstones
    for (const key of otherCRDT.tombstones) {
      this.tombstones.add(key);
      this.data.delete(key);
    }
    
    // Actualizar vector clock (tomar máximos)
    for (const [node, time] of otherCRDT.vectorClock) {
      const current = this.vectorClock.get(node) || 0;
      this.vectorClock.set(node, Math.max(current, time));
    }
    
    return conflicts; // Para logging/debug
  }

  resolveConflict(a, b) {
    if (a.timestamp !== b.timestamp) {
      return a.timestamp > b.timestamp ? 'ours' : 'theirs';
    }
    // Desempate determinístico por nodeId (lexicográfico)
    return a.nodeId > b.nodeId ? 'ours' : 'theirs';
  }

  // Serializar para transmisión (Bluetooth/WiFi Direct)
  serialize() {
    return JSON.stringify({
      nodeId: this.nodeId,
      vectorClock: Array.from(this.vectorClock),
      data: Array.from(this.data),
      tombstones: Array.from(this.tombstones)
    });
  }

  static deserialize(json) {
    const parsed = JSON.parse(json);
    const crdt = new GenotypeCRDT(parsed.nodeId);
    crdt.vectorClock = new Map(parsed.vectorClock);
    crdt.data = new Map(parsed.data);
    crdt.tombstones = new Set(parsed.tombstones);
    return crdt;
  }
}

// Uso en sincronización rural
class RuralSyncManager {
  constructor() {
    this.crdt = new GenotypeCRDT(this.getDeviceId());
    this.peers = new Map();
  }

  getDeviceId() {
    // Generar ID único persistente por dispositivo
    let id = localStorage.getItem('genotype-device-id');
    if (!id) {
      id = 'rural-' + Math.random().toString(36).substr(2, 9);
      localStorage.setItem('genotype-device-id', id);
    }
    return id;
  }

  // Recibir datos de otro dispositivo (vía WebRTC DataChannel o Bluetooth)
  receiveSync(peerId, data) {
    const peerCRDT = GenotypeCRDT.deserialize(data);
    const conflicts = this.crdt.merge(peerCRDT);
    
    if (conflicts.length > 0) {
      console.log('[SYNC] Conflictos resueltos:', conflicts);
    }
    
    this.persistToIndexedDB();
  }

  async persistToIndexedDB() {
    const db = await openDB('genotype-sync', 1);
    await db.put('crdt', {
      id: 'main',
      data: this.crdt.serialize(),
      timestamp: Date.now()
    });
  }
}
```

### VECTOR 4: EFICIENCIA ENERGÉTICA LÍMITE (Thermal Throttling Preventivo)

**La fisura:** El iPad A10 bajo carga sostenida activa thermal throttling del CPU, degradando rendimiento 40-60%. Necesitamos predecir y prevenir antes de que ocurra.

**Solución - Modelo Predictivo de Temperatura:**
```javascript
/**
 * GENOTIPO_THERMAL_AI: Predicción de throttling basada en patrones
 * Safari 14 no expone temperatura, pero podemos inferir de timing jitter
 */

class ThermalPredictor {
  constructor() {
    this.samples = new Float64Array(300); // 5 minutos a 1Hz
    this.index = 0;
    this.model = {
      baselineFPS: 60,
      thermalInertia: 0, // 0-1, acumulación de calor
      coolingRate: 0.05,
      heatingRate: 0.1
    };
    
    this.startMonitoring();
  }

  startMonitoring() {
    // Muestrear FPS real cada segundo
    setInterval(() => this.sampleFPS(), 1000);
  }

  sampleFPS() {
    const now = performance.now();
    const delta = now - (this.lastSample || now);
    this.lastSample = now;
    
    // Calcular FPS efectivo basado en RAF
    const fps = 1000 / delta;
    this.samples[this.index] = fps;
    this.index = (this.index + 1) % 300;
    
    this.updateThermalModel(fps);
  }

  updateThermalModel(currentFPS) {
    const expectedFPS = this.model.baselineFPS;
    const drop = Math.max(0, expectedFPS - currentFPS);
    
    // Si hay drop, acumular calor. Si no, disipar
    if (drop > 5) {
      this.model.thermalInertia += this.model.heatingRate;
    } else {
      this.model.thermalInertia = Math.max(0, 
        this.model.thermalInertia - this.model.coolingRate
      );
    }
    
    // Predecir throttling inminente
    if (this.model.thermalInertia > 0.7) {
      this.triggerPreventiveMeasures();
    }
    
    // Ajustar baseline si el dispositivo ya está throttled persistentemente
    if (this.model.thermalInertia > 0.9 && currentFPS < 45) {
      this.model.baselineFPS = Math.min(this.model.baselineFPS, currentFPS);
    }
  }

  triggerPreventiveMeasures() {
    // Medidas drásticas antes de que el OS intervenga
    window.dispatchEvent(new CustomEvent('thermal-prediction', {
      detail: {
        risk: 'HIGH',
        suggestedActions: [
          'reduce-refresh-rate',
          'pause-background-sync',
          'disable-animations',
          'lower-render-quality'
        ]
      }
    }));
  }

  getThermalState() {
    return {
      inertia: this.model.thermalInertia,
      baselineFPS: this.model.baselineFPS,
      trend: this.calculateTrend(),
      riskLevel: this.model.thermalInertia > 0.7 ? 'CRITICAL' : 
                  this.model.thermalInertia > 0.4 ? 'ELEVATED' : 'NORMAL'
    };
  }

  calculateTrend() {
    // Regresión lineal simple de últimos 30 segundos
    const recent = [];
    for (let i = 0; i < 30; i++) {
      const idx = (this.index - 1 - i + 300) % 300;
      recent.push(this.samples[idx]);
    }
    
    const n = recent.length;
    const sumX = recent.reduce((a, b, i) => a + i, 0);
    const sumY = recent.reduce((a, b) => a + b, 0);
    const sumXY = recent.reduce((a, b, i) => a + i * b, 0);
    const sumXX = recent.reduce((a, b, i) => a + i * i, 0);
    
    const slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX);
    return slope; // Negativo = degradándose
  }
}

// Integración con Virtual Scroller
window.addEventListener('thermal-prediction', (e) => {
  if (e.detail.risk === 'HIGH') {
    // Reducir a 15fps efectivos (cada 4 frames)
    window.GENOTYPE_THROTTLE = 4;
    
    // Desactivar will-change para ahorrar GPU
    document.querySelectorAll('.genotype-node').forEach(node => {
      node.style.willChange = 'auto';
    });
  }
});
```

### VECTOR 5: STORAGE QUOTA EXHAUSTIÓN SILENCIOSA

**La fisura:** Safari 14 tiene límite de ~1GB por origen. Cuando se alcanza, IndexedDB empieza a fallar silenciosamente o con errores no manejados.

**Solución - Gestor de Cuotas con Evicción Inteligente:**
```javascript
class QuotaManager {
  constructor() {
    this.warningThreshold = 0.8; // 80%
    this.criticalThreshold = 0.95; // 95%
    this.checkInterval = 60000; // 1 minuto
  }

  async checkQuota() {
    if ('storage' in navigator && 'estimate' in navigator.storage) {
      const estimate = await navigator.storage.estimate();
      const usage = estimate.usage || 0;
      const quota = estimate.quota || Infinity;
      const ratio = usage / quota;
      
      if (ratio > this.criticalThreshold) {
        await this.emergencyEviction();
      } else if (ratio > this.warningThreshold) {
        await this.gentleEviction();
      }
      
      return { usage, quota, ratio };
    }
    
    // Fallback: estimar basado en tamaño de IDB
    return this.estimateFallback();
  }

  async gentleEviction() {
    // Eliminar chunks menos accesados (LRU)
    const db = await openDB('genotype-chunks', 1);
    const allChunks = await db.getAll('chunks');
    
    // Ordenar por último acceso
    allChunks.sort((a, b) => (a.lastAccessed || 0) - (b.lastAccessed || 0));
    
    // Eliminar el 10% más antiguo
    const toDelete = Math.ceil(allChunks.length * 0.1);
    const tx = db.transaction('chunks', 'readwrite');
    
    for (let i = 0; i < toDelete; i++) {
      await tx.objectStore('chunks').delete(allChunks[i].id);
    }
    
    console.log(`[QUOTA] Evicción suave: ${toDelete} chunks eliminados`);
  }

  async emergencyEviction() {
    // Conservar solo chunks críticos (índice 0-10)
    const db = await openDB('genotype-chunks', 1);
    const allKeys = await db.getAllKeys('chunks');
    
    const tx = db.transaction('chunks', 'readwrite');
    let deleted = 0;
    
    for (const key of allKeys) {
      if (key > 10) { // Mantener solo primeros 10 chunks
        await tx.objectStore('chunks').delete(key);
        deleted++;
      }
    }
    
    // Limpiar Cache API también
    const cacheNames = await caches.keys();
    for (const name of cacheNames) {
      const cache = await caches.open(name);
      const requests = await cache.keys();
      // Mantener solo assets críticos
      for (const request of requests) {
        if (!request.url.includes('critical')) {
          await cache.delete(request);
        }
      }
    }
    
    console.log(`[QUOTA] Evicción EMERGENCIA: ${deleted} chunks eliminados`);
  }

  async estimateFallback() {
    // Estimación basada en conteo de objetos
    const db = await openDB('genotype-chunks', 1);
    const chunks = await db.getAll('chunks');
    const estimatedBytes = chunks.reduce((sum, chunk) => {
      return sum + JSON.stringify(chunk).length * 2; // UTF-16 aprox
    }, 0);
    
    // Asumir 1GB límite
    return {
      usage: estimatedBytes,
      quota: 1073741824,
      ratio: estimatedBytes / 1073741824
    };
  }
}
```
