// src/utils/ProtectionKey.js
export const PROTECTION_LEVELS = {
  READ_ONLY: 0,      // Només visualització
  MINOR_EDITS: 1,    // Només correccions de tipografia
  FEATURE_ADD: 2,    // Afegir noves funcions sense tocar l'existent
  REWRITE_LOCKED: 99 // PROHIBIT: Reescriptura total
};

export const checkRewritePermission = (filePath, intendedAction) => {
  const fileStatus = getFileStatus(filePath);
  
  if (fileStatus.isLocked && intendedAction === 'FULL_REWRITE') {
    throw new Error(
      `🚨 PERMÍS DENEGAT: El fitxer '${filePath}' està PROTEGIT.\n` +
      `Motiu: Arquitectura tancada en equip.\n` +
      `Acció requerida: Demanar clau de permís al Consell de la Petorreta.\n` +
      `Codi d'error: CPD-LOCKED-FILE`
    );
  }
  
  return true;
};

// Marcador en el fitxer mateix (al capdamunt de cada component "tancat")
export const FILE_PROTECTION_MARKER = `

 */
`;
